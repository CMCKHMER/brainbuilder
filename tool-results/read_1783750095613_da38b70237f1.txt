     7→     7→interface Particle {
     8→     8→  x: number;
     9→     9→  y: number;
    10→    10→  vx: number;
    11→    11→  vy: number;
    12→    12→  size: number;
    13→    13→  opacity: number;
    14→    14→  hue: number;
    15→    15→  life: number;
    16→    16→  maxLife: number;

    18→    18→}
    19→    19→
    20→    20→function lerp(a: number, b: number, t: number) {
    21→    21→  return a + (b - a) * t;
    22→    22→}
    23→    23→
    24→    24→function randomRange(min: number, max: number) {
    25→    25→  return Math.random() * (max - min) + min;
    26→    26→}

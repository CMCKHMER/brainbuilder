   961→export function getRivalDialogue(characterClass: string, context: RivalDialogue['context'], turnNumber: number): { text: string } | null {
   962→  const cls = characterClass.toLowerCase();
   963→  const eligible = RIVAL_DIALOGUES.filter(d => {
   964→    if (d.speakerClass !== cls) return false;
   965→    if (d.context !== context) return false;
   966→    if (turnNumber < d.minTurn) return false;
   967→    return true;
   968→  });
   969→  if (eligible.length === 0) return null;
   970→  const dialogue = eligible[Math.floor(Math.random() * eligible.length)];
   971→  const line = dialogue.lines[Math.floor(Math.random() * dialogue.lines.length)];
   972→  return { text: line };
   973→}

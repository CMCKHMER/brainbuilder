    60→   109→   109→}
    61→   110→   110→
    62→   111→   111→function smoothNoise(x: number, z: number): number {
    63→   112→   112→  const ix = Math.floor(x);
    64→   113→   113→  const iz = Math.floor(z);
    65→   114→   114→  const fx = x - ix;
    66→   115→   115→  const fz = z - iz;
    67→   116→   116→  const a = simpleNoise(ix, iz);
    68→   117→   117→  const b = simpleNoise(ix + 1, iz);
    69→   118→   118→  const c = simpleNoise(ix, iz + 1);
    70→   119→   119→  const d = simpleNoise(ix + 1, iz + 1);
    71→   120→   120→  const ux = fx * fx * (3 - 2 * fx);
    72→   121→   121→  const uz = fz * fz * (3 - 2 * fz);
    73→   122→   122→  return a + (b - a) * ux + (c - a) * uz + (a - b - c + d) * ux * uz;
    74→   123→   123→}
    75→   124→   124→
    76→   125→   125→function fbmNoise(x: number, z: number, octaves: number = 4): number {
    77→   126→   126→  let val = 0;
    78→   127→   127→  let amp = 1;
    79→   128→   128→  let freq = 1;
    80→   129→   129→  let max = 0;
    81→   130→   130→  for (let i = 0; i < octaves; i++) {
    82→   131→   131→    val += smoothNoise(x * freq, z * freq) * amp;
    83→   132→   132→    max += amp;
    84→   133→   133→    amp *= 0.5;
    85→   134→   134→    freq *= 2;
    86→   135→   135→  }
    87→   136→   136→  return val / max;
    88→   137→   137→}

   153→   202→   202→function sampleGradient(palette: RegionPalette, t: number): [number, number, number] {
   154→   203→   203→  t = Math.max(0, Math.min(1, t));
   155→   204→   204→  if (t < 0.33) {
   156→   205→   205→    const f = t / 0.33;
   157→   206→   206→    return [
   158→   207→   207→      palette.lowColor[0] + (palette.midColor[0] - palette.lowColor[0]) * f,
   159→   208→   208→      palette.lowColor[1] + (palette.midColor[1] - palette.lowColor[1]) * f,
   160→   209→   209→      palette.lowColor[2] + (palette.midColor[2] - palette.lowColor[2]) * f,
   161→   210→   210→    ];
   162→   211→   211→  } else if (t < 0.66) {
   163→   212→   212→    const f = (t - 0.33) / 0.33;
   164→   213→   213→    return [
   165→   214→   214→      palette.midColor[0] + (palette.highColor[0] - palette.midColor[0]) * f,
   166→   215→   215→      palette.midColor[1] + (palette.highColor[1] - palette.midColor[1]) * f,
   167→   216→   216→      palette.midColor[2] + (palette.highColor[2] - palette.midColor[2]) * f,
   168→   217→   217→    ];
   169→   218→   218→  } else {
   170→   219→   219→    const f = (t - 0.66) / 0.34;
   171→   220→   220→    return [
   172→   221→   221→      palette.highColor[0] + (palette.peakColor[0] - palette.highColor[0]) * f,
   173→   222→   222→      palette.highColor[1] + (palette.peakColor[1] - palette.highColor[1]) * f,
   174→   223→   223→      palette.highColor[2] + (palette.peakColor[2] - palette.highColor[2]) * f,
   175→   224→   224→    ];
   176→   225→   225→  }
   177→   226→   226→}
   178→   227→   227→
   179→   228→   228→function createTerritoryGeometry(
   180→   229→   229→  pathStr: string,
   181→   230→   230→  baseHeight: number,
   182→   231→   231→  territoryId: string,
   183→   232→   232→  region: string,
   184→   233→   233→): THREE.BufferGeometry {
   185→   234→   234→  const points = parseSVGPath(pathStr);
   186→   235→   235→  if (points.length < 3) {
   187→   236→   236→    return new THREE.ExtrudeGeometry(new THREE.Shape(), { depth: 0.1 });
   188→   237→   237→  }
   189→   238→   238→  const shape = new THREE.Shape();
   190→   239→   239→  const [x0, z0] = svgTo3D(points[0][0], points[0][1]);
   191→   240→   240→  shape.moveTo(x0, z0);
   192→   241→   241→  for (let i = 1; i < points.length; i++) {
   193→   242→   242→    const [x, z] = svgTo3D(points[i][0], points[i][1]);
   194→   243→   243→    shape.lineTo(x, z);
   195→   244→   244→  }
   196→   245→   245→  shape.closePath();
   197→   246→   246→
   198→   247→   247→  const geo = new THREE.ExtrudeGeometry(shape, {
   199→   248→   248→    steps: 1,
   200→   249→   249→    depth: baseHeight,
   201→   250→   250→    bevelEnabled: true,
   202→   251→   251→    bevelThickness: 0.015,
   203→   252→   252→    bevelSize: 0.015,
   204→   253→   253→    bevelOffset: 0,
   205→   254→   254→    bevelSegments: 2,
   206→   255→   255→  });
   207→   256→   256→  geo.rotateX(-Math.PI / 2);
   208→   257→   257→  geo.translate(0, 0.05, 0);
   209→   258→   258→
   210→   259→   259→  // ---- Vertex displacement for hills ----
   211→   260→   260→  const posAttr = geo.attributes.position;
   212→   261→   261→  const posArr = posAttr.array as Float32Array;
   213→   262→   262→  const vertexCount = posAttr.count;
   214→   263→   263→
   215→   264→   264→  // Find the Y range to identify top face vs sides
   216→   265→   265→  let minY = Infinity, maxY = -Infinity;
   217→   266→   266→  for (let i = 0; i < vertexCount; i++) {
   218→   267→   267→    const y = posArr[i * 3 + 1];
   219→   268→   268→    if (y < minY) minY = y;
   220→   269→   269→    if (y > maxY) maxY = y;
   221→   270→   270→  }
   222→   271→   271→  const topThreshold = maxY - 0.06; // Top face vertices
   223→   272→   272→
   224→   273→   273→  // Compute centroid for distance-based height falloff
   225→   274→   274→  let cx = 0, cz = 0, count = 0;
   226→   275→   275→  for (let i = 0; i < vertexCount; i++) {
   227→   276→   276→    if (posArr[i * 3 + 1] >= topThreshold) {
   228→   277→   277→      cx += posArr[i * 3];
   229→   278→   278→      cz += posArr[i * 3 + 2];
   230→   279→   279→      count++;
   231→   280→   280→    }
   232→   281→   281→  }
   233→   282→   282→  cx /= (count || 1);
   234→   283→   283→  cz /= (count || 1);
   235→   284→   284→
   236→   285→   285→  // Compute max distance from centroid for normalization
   237→   286→   286→  let maxDist = 0;
   238→   287→   287→  for (let i = 0; i < vertexCount; i++) {
   239→   288→   288→    if (posArr[i * 3 + 1] >= topThreshold) {
   240→   289→   289→      const dx = posArr[i * 3] - cx;
   241→   290→   290→      const dz = posArr[i * 3 + 2] - cz;
   242→   291→   291→      const d = Math.sqrt(dx * dx + dz * dz);
   243→   292→   292→      if (d > maxDist) maxDist = d;
   244→   293→   293→    }
   245→   294→   294→  }
   246→   295→   295→
   247→   296→   296→  // Hill height parameters per territory (seeded by territory name hash)
   248→   297→   297→  let seed = 0;
   249→   298→   298→  for (let c = 0; c < territoryId.length; c++) seed = seed * 31 + territoryId.charCodeAt(c);
   250→   299→   299→  const hillScale = 0.15 + (Math.abs(seed) % 100) / 100 * 0.2; // 0.15 - 0.35
   251→   300→   300→  const noiseScale = 3 + (Math.abs(seed * 7) % 100) / 100 * 4; // 3 - 7
   252→   301→   301→  const ridgeFactor = (Math.abs(seed * 13) % 100) / 100; // 0-1
   253→   302→   302→
   254→   303→   303→  // Displace top face vertices
   255→   304→   304→  const heights: number[] = new Array(vertexCount).fill(0);
   256→   305→   305→  for (let i = 0; i < vertexCount; i++) {
   257→   306→   306→    if (posArr[i * 3 + 1] >= topThreshold) {
   258→   307→   307→      const x = posArr[i * 3];
   259→   308→   308→      const z = posArr[i * 3 + 2];
   260→   309→   309→      const dx = x - cx;
   261→   310→   310→      const dz = z - cz;
   262→   311→   311→      const dist = Math.sqrt(dx * dx + dz * dz);
   263→   312→   312→      const distNorm = maxDist > 0 ? dist / maxDist : 0;
   264→   313→   313→
   265→   314→   314→      // Distance falloff: higher near center, lower at edges
   266→   315→   315→      const falloff = Math.max(0, 1 - distNorm * distNorm);
   267→   316→   316→
   268→   317→   317→      // Multi-octave noise for natural terrain
   269→   318→   318→      const n1 = fbmNoise(x * noiseScale + seed, z * noiseScale + seed * 0.7, 4);
   270→   319→   319→      const n2 = fbmNoise(x * noiseScale * 0.5 + seed * 2, z * noiseScale * 0.5 + seed, 3);
   271→   320→   320→
   272→   321→   321→      // Ridge noise: creates mountain ridges
   273→   322→   322→      const ridge = 1 - Math.abs(n1 * 2 - 1);
   274→   323→   323→      const heightVal = (n1 * 0.5 + n2 * 0.3 + ridge * ridgeFactor * 0.4) * hillScale * falloff;
   275→   324→   324→
   276→   325→   325→      posArr[i * 3 + 1] += heightVal;
   277→   326→   326→      heights[i] = heightVal;
   278→   327→   327→    }
   279→   328→   328→  }
   280→   329→   329→  posAttr.needsUpdate = true;
   281→   330→   330→  geo.computeVertexNormals();
   282→   331→   331→
   283→   332→   332→  // ---- Vertex colors for gradient shading ----
   284→   333→   333→  const palette = getRegionPalette(region);
   285→   334→   334→  const colors = new Float32Array(vertexCount * 3);
   286→   335→   335→
   287→   336→   336→  // Find the max height for normalization
   288→   337→   337→  let maxH = 0;
   289→   338→   338→  for (let i = 0; i < vertexCount; i++) {
   290→   339→   339→    if (heights[i] > maxH) maxH = heights[i];
   291→   340→   340→  }
   292→   341→   341→
   293→   342→   342→  for (let i = 0; i < vertexCount; i++) {
   294→   343→   343→    const y = posArr[i * 3 + 1];
   295→   344→   344→    const isTop = y >= topThreshold - 0.01;
   296→   345→   345→
   297→   346→   346→    if (isTop && maxH > 0) {
   298→   347→   347→      const t = heights[i] / maxH;
   299→   348→   348→      const [r, g, b] = sampleGradient(palette, t);
   300→   349→   349→      colors[i * 3] = r;
   301→   350→   350→      colors[i * 3 + 1] = g;
   302→   351→   351→      colors[i * 3 + 2] = b;
   303→   352→   352→    } else {
   304→   353→   353→      // Side faces: darken version of the low color
   305→   354→   354→      colors[i * 3] = palette.lowColor[0] * 0.6;
   306→   355→   355→      colors[i * 3 + 1] = palette.lowColor[1] * 0.6;
   307→   356→   356→      colors[i * 3 + 2] = palette.lowColor[2] * 0.6;
   308→   357→   357→    }
   309→   358→   358→  }
   310→   359→   359→
   311→   360→   360→  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   312→   361→   361→  return geo;
   313→   362→   362→}

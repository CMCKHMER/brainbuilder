    17→  if (_webGLSupported !== null) return _webGLSupported;
    18→  if (typeof window === 'undefined') { _webGLSupported = false; return false; }
    19→  try {
    20→    const canvas = document.createElement('canvas');

    22→    _webGLSupported = !!gl;
    23→  } catch {
    24→    _webGLSupported = false;
    25→  }
    26→  return _webGLSupported;

    72→function parseSVGPath(pathStr: string): [number, number][] {
    73→  const points: [number, number][] = [];
    74→  const parts = pathStr.trim().split(/\s+/);
    75→  let i = 0;
    76→  while (i < parts.length) {
    77→    const cmd = parts[i];
    78→    if (cmd === 'M' || cmd === 'L') {
    79→      const x = parseFloat(parts[i + 1]);
    80→      const y = parseFloat(parts[i + 2]);
    81→      if (!isNaN(x) && !isNaN(y)) {
    82→        points.push([x, y]);
    83→      }
    84→      i += 3;
    85→    } else if (cmd === 'Z') {
    86→      break;
    87→    } else {
    88→      i++;
    89→    }
    90→  }
    91→  return points;
    92→}
    93→
    94→function svgTo3D(sx: number, sy: number): [number, number] {
    95→  return [(sx - CENTER_X) * SCALE, -(sy - CENTER_Y) * SCALE];
    96→}
    97→
    98→function createTerritoryGeometry(pathStr: string, height: number): THREE.ExtrudeGeometry {
    99→  const points = parseSVGPath(pathStr);
   100→  if (points.length < 3) {
   101→    return new THREE.ExtrudeGeometry(new THREE.Shape(), { depth: 0.1 });
   102→  }
   103→  const shape = new THREE.Shape();
   104→  const [x0, z0] = svgTo3D(points[0][0], points[0][1]);
   105→  shape.moveTo(x0, z0);
   106→  for (let i = 1; i < points.length; i++) {
   107→    const [x, z] = svgTo3D(points[i][0], points[i][1]);
   108→    shape.lineTo(x, z);
   109→  }
   110→  shape.closePath();
   111→
   112→  const geo = new THREE.ExtrudeGeometry(shape, {
   113→    steps: 1,
   114→    depth: height,
   115→    bevelEnabled: true,
   116→    bevelThickness: 0.015,
   117→    bevelSize: 0.015,
   118→    bevelOffset: 0,
   119→    bevelSegments: 2,
   120→  });
   121→  geo.rotateX(-Math.PI / 2);
   122→  geo.translate(0, 0.05, 0);
   123→  return geo;
   124→}

   511→function TerritoryMesh({ territoryId, path, labelX, labelY, onHover, onClick }: TerritoryMeshProps) {
   512→  const territories = useGameStore(s => s.territories);
   513→  const selectedTerritory = useGameStore(s => s.selectedTerritory);
   514→  const targetTerritory = useGameStore(s => s.targetTerritory);
   515→  const phase = useGameStore(s => s.phase);
   516→  const players = useGameStore(s => s.players);
   517→  const reinforcementsLeft = useGameStore(s => s.reinforcementsLeft);
   518→  const deployUnitType = useGameStore(s => s.deployUnitType);
   519→  const currentPlayerIndex = useGameStore(s => s.currentPlayerIndex);
   520→  const selectTerritory = useGameStore(s => s.selectTerritory);
   521→  const deployArmy = useGameStore(s => s.deployArmy);
   522→
   523→  const territory = territories[territoryId];
   524→  if (!territory) return null;
   525→
   526→  const visual = TERRITORY_VISUAL[territoryId] || { structure: 'tower' as StructureType, height: 0.2, structureColor: '#8B7355', terrainColor: '#5A5A3A', terrainSideColor: '#3A3A2A' };
   527→  const playerColorMap = useMemo(() => {
   528→    const map: Record<string, { color: string; colorLight: string }> = {};
   529→    players.forEach(p => { map[p.id] = { color: p.color, colorLight: p.colorLight }; });
   530→    return map;
   531→  }, [players]);
   532→
   533→  const owner = territory.ownerId ? playerColorMap[territory.ownerId] : null;
   534→  const currentPlayerId = phase !== 'setup' && phase !== 'gameover' ? players[currentPlayerIndex]?.id : null;
   535→
   536→  const isSelected = territoryId === selectedTerritory;
   537→  const isTarget = territoryId === targetTerritory;
   538→
   539→  const geometry = useMemo(() => createTerritoryGeometry(path, visual.height), [path, visual.height]);
   540→  const edgesGeometry = useMemo(() => new THREE.EdgesGeometry(geometry, 25), [geometry]);
   541→
   542→  const [cx, cz] = svgTo3D(labelX, labelY);
   543→  const structY = 0.05 + visual.height;
   544→  const dominant = getDominantUnit(territory.units);
   545→
   546→  const isSelectable = useMemo(() => {
   547→    if (phase === 'deploy') {
   548→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   549→      return territory.ownerId === currentPlayerId && reinforcementsLeft >= cost;
   550→    }
   551→    if (phase === 'attack') {
   552→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   553→      if (territoryId === selectedTerritory) return true;
   554→      if (territory.ownerId !== currentPlayerId) return territory.adjacentTo.includes(selectedTerritory);
   555→      return territory.ownerId === currentPlayerId && territory.units.length > 1;
   556→    }
   557→    if (phase === 'fortify') {
   558→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   559→      if (territoryId === selectedTerritory) return true;
   560→      if (territory.ownerId === currentPlayerId && selectedTerritory !== null) return territory.adjacentTo.includes(selectedTerritory);
   561→      return false;
   562→    }
   563→    return false;
   564→  }, [phase, territory, selectedTerritory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex]);
   565→
   566→  const topColor = owner ? owner.colorLight : visual.terrainColor;
   567→  const sideColor = owner ? owner.color + 'AA' : visual.terrainSideColor;
   568→
   569→  const emissiveColor = isSelected ? '#FFD700' : isTarget ? '#FF0000' : '#000000';
   570→
   571→  const meshRef = useRef<THREE.Mesh>(null);
   572→  const edgesRef = useRef<THREE.LineSegments>(null);
   573→
   574→  useFrame((state) => {
   575→    if (meshRef.current) {
   576→      const mat = meshRef.current.material as THREE.MeshStandardMaterial;
   577→      if (isSelected) {
   578→        mat.emissiveIntensity = 0.3 + Math.sin(state.clock.elapsedTime * 3) * 0.15;
   579→      } else if (isTarget) {
   580→        mat.emissiveIntensity = 0.4 + Math.sin(state.clock.elapsedTime * 4) * 0.2;
   581→      } else {
   582→        mat.emissiveIntensity = 0;
   583→      }
   584→    }
   585→    if (edgesRef.current) {
   586→      const mat = edgesRef.current.material as THREE.LineBasicMaterial;
   587→      if (isSelected) {
   588→        mat.color.set('#FFD700');
   589→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 3) * 0.3;
   590→      } else if (isTarget) {
   591→        mat.color.set('#FF0000');
   592→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 4) * 0.3;
   593→      } else {
   594→        mat.color.set('#3D2B1F');
   595→        mat.opacity = 0.8;
   596→      }
   597→    }
   598→  });
   599→
   600→  const handleClick = useCallback((e: ThreeEvent<MouseEvent>) => {
   601→    e.stopPropagation();
   602→    if (phase === 'deploy') {
   603→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   604→      if (territory.ownerId === currentPlayerId && reinforcementsLeft >= cost) {
   605→        deployArmy(territoryId, deployUnitType);
   606→      }
   607→    } else if (phase === 'attack' || phase === 'fortify') {
   608→      selectTerritory(territoryId);
   609→    }
   610→  }, [phase, territory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex, selectTerritory, deployArmy, territoryId]);
   611→
   612→  const handlePointerOver = useCallback((e: ThreeEvent<PointerEvent>) => {
   613→    e.stopPropagation();
   614→    if (isSelectable) {
   615→      document.body.style.cursor = 'pointer';
   616→    }
   617→    onHover(territoryId);
   618→  }, [isSelectable, territoryId, onHover]);
   619→
   620→  const handlePointerOut = useCallback((e: ThreeEvent<PointerEvent>) => {
   621→    e.stopPropagation();
   622→    document.body.style.cursor = 'auto';
   623→    onHover(null);
   624→  }, [onHover]);
   625→
   626→  return (
   627→    <group>
   628→      {/* Territory extruded shape */}
   629→      <mesh
   630→        ref={meshRef}
   631→        geometry={geometry}
   632→        onClick={handleClick}
   633→        onPointerOver={handlePointerOver}
   634→        onPointerOut={handlePointerOut}
   635→      >
   636→        <meshStandardMaterial
   637→          color={topColor}
   638→          roughness={0.75}
   639→          metalness={0.05}
   640→          emissive={emissiveColor}
   641→          emissiveIntensity={0}
   642→          transparent={!isSelectable}
   643→          opacity={isSelectable ? 1 : (phase === 'attack' || phase === 'fortify') ? 0.6 : 0.85}
   644→        />
   645→      </mesh>
   646→
   647→      {/* Territory border edges */}
   648→      <lineSegments ref={edgesRef} geometry={edgesGeometry}>
   649→        <lineBasicMaterial color="#3D2B1F" transparent opacity={0.8} />
   650→      </lineSegments>
   651→
   652→      {/* 3D Structure - positioned so base sits on top of territory */}
   653→      <StructureRenderer type={visual.structure} position={[cx, structY, cz]} color={visual.structureColor} />
   654→
   655→      {/* Territory label and unit badge */}
   656→      <Html position={[cx, structY + (visual.structure === 'mountain' ? 2.8 : visual.structure === 'tower' ? 3.0 : visual.structure === 'fortress' ? 1.8 : visual.structure === 'forest' ? 1.2 : 1.5), cz]} center>
   657→        <div style={{ pointerEvents: 'none', userSelect: 'none', textAlign: 'center' }}>
   658→          <div style={{
   659→            fontSize: '9px',
   660→            fontFamily: 'var(--font-cinzel), serif',
   661→            color: 'rgba(255,255,255,0.9)',
   662→            textShadow: '0 1px 4px rgba(0,0,0,0.9), 0 0 8px rgba(0,0,0,0.6)',
   663→            whiteSpace: 'nowrap',
   664→            letterSpacing: '1px',
   665→          }}>
   666→            {territory.name}
   667→          </div>
   668→          {territory.units.length > 0 && dominant && (
   669→            <div style={{
   670→              marginTop: '2px',
   671→              display: 'inline-flex',
   672→              alignItems: 'center',
   673→              gap: '2px',
   674→              background: 'rgba(0,0,0,0.6)',
   675→              borderRadius: '8px',
   676→              padding: '1px 5px',
   677→              border: `1px solid ${owner ? owner.color + '66' : 'rgba(139,115,85,0.3)'}`,
   678→            }}>
   679→              <span style={{ fontSize: '9px' }}>{UNIT_TYPES[dominant].icon}</span>
   680→              <span style={{
   681→                fontSize: '8px',
   682→                fontWeight: 'bold',
   683→                color: owner ? owner.colorLight : '#ccc',
   684→                textShadow: '0 1px 2px rgba(0,0,0,0.8)',
   685→              }}>
   686→                {territory.units.length}
   687→              </span>
   688→            </div>
   689→          )}
   690→        </div>
   691→      </Html>
   692→    </group>
   693→  );
   694→}

   697→// Connection Lines

   699→function ConnectionLines() {
   700→  const territories = useGameStore(s => s.territories);
   701→
   702→  const lines = useMemo(() => {
   703→    const result: { start: [number, number, number]; end: [number, number, number] }[] = [];
   704→    const drawn = new Set<string>();
   705→    for (const t of Object.values(territories)) {
   706→      const vis = TERRITORY_VISUAL[t.id];
   707→      const h = (vis?.height || 0.2) + 0.1;
   708→      for (const adjId of t.adjacentTo) {
   709→        const key = [t.id, adjId].sort().join('-');
   710→        if (!drawn.has(key)) {
   711→          drawn.add(key);
   712→          const adj = territories[adjId];
   713→          if (adj) {
   714→            const [x1, z1] = svgTo3D(t.labelX, t.labelY);
   715→            const [x2, z2] = svgTo3D(adj.labelX, adj.labelY);
   716→            const adjVis = TERRITORY_VISUAL[adjId];
   717→            const h2 = (adjVis?.height || 0.2) + 0.1;
   718→            const avgH = (h + h2) / 2 + 0.08;
   719→            result.push({ start: [x1, avgH, z1], end: [x2, avgH, z2] });
   720→          }
   721→        }
   722→      }
   723→    }
   724→    return result;
   725→  }, [territories]);
   726→
   727→  return (
   728→    <group>
   729→      {lines.map((line, i) => (
   730→        <Line
   731→          key={`conn-${i}`}
   732→          points={[line.start, line.end]}
   733→          color="#8B735544"
   734→          lineWidth={1}
   735→          dashed
   736→          dashSize={0.08}
   737→          gapSize={0.08}
   738→        />
   739→      ))}
   740→    </group>
   741→  );
   742→}

  1026→function Fallback2DMap({ onTerritoryHover }: { onTerritoryHover: (id: string | null) => void }) {
  1027→  const territories = useGameStore(s => s.territories);
  1028→  const players = useGameStore(s => s.players);
  1029→  const selectedTerritory = useGameStore(s => s.selectedTerritory);
  1030→  const selectTerritory = useGameStore(s => s.selectTerritory);
  1031→  const phase = useGameStore(s => s.phase);
  1032→
  1033→  return (
  1034→    <div
  1035→      className="w-full h-full relative overflow-hidden"
  1036→      style={{ background: 'linear-gradient(180deg, #0a1520 0%, #0d1a12 50%, #1a1510 100%)' }}
  1037→    >
  1038→      {/* Ocean background */}
  1039→      <div
  1040→        className="absolute inset-0"
  1041→        style={{
  1042→          background: 'radial-gradient(ellipse at 50% 50%, #0a1a2e 0%, #060810 70%)',
  1043→        }}
  1044→      />
  1045→
  1046→      {/* SVG Map */}
  1047→      <svg
  1048→        viewBox="0 0 1000 650"
  1049→        className="w-full h-full"
  1050→        style={{ maxHeight: '100%' }}
  1051→      >
  1052→        {TERRITORY_DEFS.map(tDef => {
  1053→          const tState = territories[tDef.id];
  1054→          if (!tState) return null;
  1055→          const owner = tState.ownerId ? players.find(p => p.id === tState.ownerId) : null;
  1056→          const isSelected = selectedTerritory === tDef.id;
  1057→          const unitCount = tState.units.length;
  1058→
  1059→          return (
  1060→            <g
  1061→              key={tDef.id}
  1062→              onClick={() => (phase === 'attack' || phase === 'fortify' || phase === 'deploy') && selectTerritory(tDef.id)}
  1063→              onMouseEnter={() => onTerritoryHover(tDef.id)}
  1064→              onMouseLeave={() => onTerritoryHover(null)}
  1065→              style={{ cursor: 'pointer' }}
  1066→            >
  1067→              <path
  1068→                d={tDef.path}
  1069→                fill={owner ? owner.color + '44' : 'rgba(60,50,40,0.3)'}
  1070→                stroke={isSelected ? '#D4A017' : owner ? owner.color + '88' : 'rgba(139,115,85,0.3)'}
  1071→                strokeWidth={isSelected ? 2.5 : 1.2}
  1072→                style={{ transition: 'fill 0.2s, stroke 0.2s' }}
  1073→              />
  1074→              {/* Territory label */}
  1075→              <text
  1076→                x={tDef.labelX}
  1077→                y={tDef.labelY}
  1078→                textAnchor="middle"
  1079→                dominantBaseline="middle"
  1080→                style={{
  1081→                  fontFamily: 'var(--font-cinzel), serif',
  1082→                  fontSize: '9px',
  1083→                  fill: owner ? owner.colorLight : '#8B7355',
  1084→                  pointerEvents: 'none',
  1085→                  textShadow: '0 1px 3px rgba(0,0,0,0.9)',
  1086→                }}
  1087→              >
  1088→                {tDef.name}
  1089→              </text>
  1090→              {/* Unit count */}
  1091→              {unitCount > 0 && (
  1092→                <text
  1093→                  x={tDef.labelX}
  1094→                  y={tDef.labelY + 14}
  1095→                  textAnchor="middle"
  1096→                  dominantBaseline="middle"
  1097→                  style={{
  1098→                    fontFamily: 'var(--font-cinzel), serif',
  1099→                    fontSize: '8px',
  1100→                    fill: owner ? owner.color : '#666',
  1101→                    pointerEvents: 'none',
  1102→                  }}
  1103→                >
  1104→                  {unitCount} ⚔️
  1105→                </text>
  1106→              )}
  1107→            </g>
  1108→          );
  1109→        })}
  1110→      </svg>
  1111→
  1112→      {/* WebGL warning banner */}
  1113→      <div
  1114→        className="absolute bottom-2 left-2 px-3 py-1.5 rounded text-[9px]"
  1115→        style={{
  1116→          background: 'rgba(0,0,0,0.7)',
  1117→          border: '1px solid rgba(220,38,38,0.3)',
  1118→          color: '#EF4444',
  1119→          fontFamily: 'var(--font-cinzel), serif',
  1120→        }}
  1121→      >
  1122→        ⚠️ 3D Unavailable — Showing 2D Map
  1123→      </div>
  1124→    </div>
  1125→  );
  1126→}

  1132→interface GameMap3DProps {
  1133→  onTerritoryHover?: (id: string | null) => void;
  1134→}
  1135→
  1136→export default function GameMap3D({ onTerritoryHover = () => {} }: GameMap3DProps) {
  1137→  const [webGL, setWebGL] = useState<boolean | null>(null);
  1138→  const handleHover = useCallback((id: string | null) => {
  1139→    onTerritoryHover(id);
  1140→  }, [onTerritoryHover]);
  1141→
  1142→  useEffect(() => {
  1143→    setWebGL(isWebGLSupported());
  1144→  }, []);
  1145→
  1146→  // Still detecting
  1147→  if (webGL === null) return <LoadingScreen />;
  1148→
  1149→  // WebGL not available — show 2D fallback
  1150→  if (!webGL) return <Fallback2DMap onTerritoryHover={handleHover} />;
  1151→
  1152→  return (
  1153→    <Suspense fallback={<LoadingScreen />}>
  1154→      <Canvas
  1155→        camera={{ position: [0, 13, 12], fov: 62, near: 0.1, far: 200 }}
  1156→        gl={{ antialias: true, alpha: false, powerPreference: 'high-performance' }}
  1157→        style={{ background: '#060810' }}
  1158→        onPointerMissed={() => {
  1159→          useGameStore.getState().clearSelection();
  1160→        }}
  1161→        onError={() => {
  1162→          // If Canvas still fails after detection, show fallback
  1163→          setWebGL(false);
  1164→        }}
  1165→      >
  1166→        <WorldScene onTerritoryHover={handleHover} />
  1167→      </Canvas>
  1168→    </Suspense>
  1169→  );
  1170→}

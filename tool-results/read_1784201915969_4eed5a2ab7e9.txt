    72→    72→function parseSVGPath(pathStr: string): [number, number][] {
    73→    73→  const points: [number, number][] = [];
    74→    74→  const parts = pathStr.trim().split(/\s+/);
    75→    75→  let i = 0;
    76→    76→  while (i < parts.length) {
    77→    77→    const cmd = parts[i];
    78→    78→    if (cmd === 'M' || cmd === 'L') {
    79→    79→      const x = parseFloat(parts[i + 1]);
    80→    80→      const y = parseFloat(parts[i + 2]);
    81→    81→      if (!isNaN(x) && !isNaN(y)) {
    82→    82→        points.push([x, y]);
    83→    83→      }
    84→    84→      i += 3;
    85→    85→    } else if (cmd === 'Z') {
    86→    86→      break;
    87→    87→    } else {
    88→    88→      i++;
    89→    89→    }
    90→    90→  }
    91→    91→  return points;
    92→    92→}
    93→    93→
    94→    94→function svgTo3D(sx: number, sy: number): [number, number] {
    95→    95→  return [(sx - CENTER_X) * SCALE, -(sy - CENTER_Y) * SCALE];
    96→    96→}

    98→    98→function createTerritoryGeometry(pathStr: string, height: number): THREE.ExtrudeGeometry {
    99→    99→  const points = parseSVGPath(pathStr);
   100→   100→  if (points.length < 3) {
   101→   101→    return new THREE.ExtrudeGeometry(new THREE.Shape(), { depth: 0.1 });
   102→   102→  }
   103→   103→  const shape = new THREE.Shape();
   104→   104→  const [x0, z0] = svgTo3D(points[0][0], points[0][1]);
   105→   105→  shape.moveTo(x0, z0);
   106→   106→  for (let i = 1; i < points.length; i++) {
   107→   107→    const [x, z] = svgTo3D(points[i][0], points[i][1]);
   108→   108→    shape.lineTo(x, z);
   109→   109→  }
   110→   110→  shape.closePath();
   111→   111→
   112→   112→  const geo = new THREE.ExtrudeGeometry(shape, {
   113→   113→    steps: 1,
   114→   114→    depth: height,
   115→   115→    bevelEnabled: true,
   116→   116→    bevelThickness: 0.015,
   117→   117→    bevelSize: 0.015,
   118→   118→    bevelOffset: 0,
   119→   119→    bevelSegments: 2,
   120→   120→  });
   121→   121→  geo.rotateX(-Math.PI / 2);
   122→   122→  geo.translate(0, 0.05, 0);
   123→   123→  return geo;
   124→   124→}

   511→   511→function TerritoryMesh({ territoryId, path, labelX, labelY, onHover, onClick }: TerritoryMeshProps) {
   512→   512→  const territories = useGameStore(s => s.territories);
   513→   513→  const selectedTerritory = useGameStore(s => s.selectedTerritory);
   514→   514→  const targetTerritory = useGameStore(s => s.targetTerritory);
   515→   515→  const phase = useGameStore(s => s.phase);
   516→   516→  const players = useGameStore(s => s.players);
   517→   517→  const reinforcementsLeft = useGameStore(s => s.reinforcementsLeft);
   518→   518→  const deployUnitType = useGameStore(s => s.deployUnitType);
   519→   519→  const currentPlayerIndex = useGameStore(s => s.currentPlayerIndex);
   520→   520→  const selectTerritory = useGameStore(s => s.selectTerritory);
   521→   521→  const deployArmy = useGameStore(s => s.deployArmy);
   522→   522→
   523→   523→  const territory = territories[territoryId];
   524→   524→  if (!territory) return null;
   525→   525→
   526→   526→  const visual = TERRITORY_VISUAL[territoryId] || { structure: 'tower' as StructureType, height: 0.2, structureColor: '#8B7355', terrainColor: '#5A5A3A', terrainSideColor: '#3A3A2A' };
   527→   527→  const playerColorMap = useMemo(() => {
   528→   528→    const map: Record<string, { color: string; colorLight: string }> = {};
   529→   529→    players.forEach(p => { map[p.id] = { color: p.color, colorLight: p.colorLight }; });
   530→   530→    return map;
   531→   531→  }, [players]);
   532→   532→
   533→   533→  const owner = territory.ownerId ? playerColorMap[territory.ownerId] : null;
   534→   534→  const currentPlayerId = phase !== 'setup' && phase !== 'gameover' ? players[currentPlayerIndex]?.id : null;
   535→   535→
   536→   536→  const isSelected = territoryId === selectedTerritory;
   537→   537→  const isTarget = territoryId === targetTerritory;
   538→   538→
   539→   539→  const geometry = useMemo(() => createTerritoryGeometry(path, visual.height), [path, visual.height]);
   540→   540→  const edgesGeometry = useMemo(() => new THREE.EdgesGeometry(geometry, 25), [geometry]);
   541→   541→
   542→   542→  const [cx, cz] = svgTo3D(labelX, labelY);
   543→   543→  const structY = 0.05 + visual.height;
   544→   544→  const dominant = getDominantUnit(territory.units);
   545→   545→
   546→   546→  const isSelectable = useMemo(() => {
   547→   547→    if (phase === 'deploy') {
   548→   548→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   549→   549→      return territory.ownerId === currentPlayerId && reinforcementsLeft >= cost;
   550→   550→    }
   551→   551→    if (phase === 'attack') {
   552→   552→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   553→   553→      if (territoryId === selectedTerritory) return true;
   554→   554→      if (territory.ownerId !== currentPlayerId) return territory.adjacentTo.includes(selectedTerritory);
   555→   555→      return territory.ownerId === currentPlayerId && territory.units.length > 1;
   556→   556→    }
   557→   557→    if (phase === 'fortify') {
   558→   558→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   559→   559→      if (territoryId === selectedTerritory) return true;
   560→   560→      if (territory.ownerId === currentPlayerId && selectedTerritory !== null) return territory.adjacentTo.includes(selectedTerritory);
   561→   561→      return false;
   562→   562→    }
   563→   563→    return false;
   564→   564→  }, [phase, territory, selectedTerritory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex]);
   565→   565→
   566→   566→  const topColor = owner ? owner.colorLight : visual.terrainColor;
   567→   567→  const sideColor = owner ? owner.color + 'AA' : visual.terrainSideColor;
   568→   568→
   569→   569→  const emissiveColor = isSelected ? '#FFD700' : isTarget ? '#FF0000' : '#000000';
   570→   570→
   571→   571→  const meshRef = useRef<THREE.Mesh>(null);
   572→   572→  const edgesRef = useRef<THREE.LineSegments>(null);
   573→   573→
   574→   574→  useFrame((state) => {
   575→   575→    if (meshRef.current) {
   576→   576→      const mat = meshRef.current.material as THREE.MeshStandardMaterial;
   577→   577→      if (isSelected) {
   578→   578→        mat.emissiveIntensity = 0.3 + Math.sin(state.clock.elapsedTime * 3) * 0.15;
   579→   579→      } else if (isTarget) {
   580→   580→        mat.emissiveIntensity = 0.4 + Math.sin(state.clock.elapsedTime * 4) * 0.2;
   581→   581→      } else {
   582→   582→        mat.emissiveIntensity = 0;
   583→   583→      }
   584→   584→    }
   585→   585→    if (edgesRef.current) {
   586→   586→      const mat = edgesRef.current.material as THREE.LineBasicMaterial;
   587→   587→      if (isSelected) {
   588→   588→        mat.color.set('#FFD700');
   589→   589→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 3) * 0.3;
   590→   590→      } else if (isTarget) {
   591→   591→        mat.color.set('#FF0000');
   592→   592→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 4) * 0.3;
   593→   593→      } else {
   594→   594→        mat.color.set('#3D2B1F');
   595→   595→        mat.opacity = 0.8;
   596→   596→      }
   597→   597→    }
   598→   598→  });
   599→   599→
   600→   600→  const handleClick = useCallback((e: ThreeEvent<MouseEvent>) => {
   601→   601→    e.stopPropagation();
   602→   602→    if (phase === 'deploy') {
   603→   603→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   604→   604→      if (territory.ownerId === currentPlayerId && reinforcementsLeft >= cost) {
   605→   605→        deployArmy(territoryId, deployUnitType);
   606→   606→      }
   607→   607→    } else if (phase === 'attack' || phase === 'fortify') {
   608→   608→      selectTerritory(territoryId);
   609→   609→    }
   610→   610→  }, [phase, territory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex, selectTerritory, deployArmy, territoryId]);
   611→   611→
   612→   612→  const handlePointerOver = useCallback((e: ThreeEvent<PointerEvent>) => {
   613→   613→    e.stopPropagation();
   614→   614→    if (isSelectable) {
   615→   615→      document.body.style.cursor = 'pointer';
   616→   616→    }
   617→   617→    onHover(territoryId);
   618→   618→  }, [isSelectable, territoryId, onHover]);
   619→   619→
   620→   620→  const handlePointerOut = useCallback((e: ThreeEvent<PointerEvent>) => {
   621→   621→    e.stopPropagation();
   622→   622→    document.body.style.cursor = 'auto';
   623→   623→    onHover(null);
   624→   624→  }, [onHover]);
   625→   625→
   626→   626→  return (
   627→   627→    <group>
   628→   628→      {/* Territory extruded shape */}
   629→   629→      <mesh
   630→   630→        ref={meshRef}
   631→   631→        geometry={geometry}
   632→   632→        onClick={handleClick}
   633→   633→        onPointerOver={handlePointerOver}
   634→   634→        onPointerOut={handlePointerOut}
   635→   635→      >
   636→   636→        <meshStandardMaterial
   637→   637→          color={topColor}
   638→   638→          roughness={0.75}
   639→   639→          metalness={0.05}
   640→   640→          emissive={emissiveColor}
   641→   641→          emissiveIntensity={0}
   642→   642→          transparent={!isSelectable}
   643→   643→          opacity={isSelectable ? 1 : (phase === 'attack' || phase === 'fortify') ? 0.6 : 0.85}
   644→   644→        />
   645→   645→      </mesh>
   646→   646→
   647→   647→      {/* Territory border edges */}
   648→   648→      <lineSegments ref={edgesRef} geometry={edgesGeometry}>
   649→   649→        <lineBasicMaterial color="#3D2B1F" transparent opacity={0.8} />
   650→   650→      </lineSegments>
   651→   651→
   652→   652→      {/* 3D Structure - positioned so base sits on top of territory */}
   653→   653→      <StructureRenderer type={visual.structure} position={[cx, structY, cz]} color={visual.structureColor} />
   654→   654→
   655→   655→      {/* Territory label and unit badge */}
   656→   656→      <Html position={[cx, structY + (visual.structure === 'mountain' ? 2.8 : visual.structure === 'tower' ? 3.0 : visual.structure === 'fortress' ? 1.8 : visual.structure === 'forest' ? 1.2 : 1.5), cz]} center>
   657→   657→        <div style={{ pointerEvents: 'none', userSelect: 'none', textAlign: 'center' }}>
   658→   658→          <div style={{
   659→   659→            fontSize: '9px',
   660→   660→            fontFamily: 'var(--font-cinzel), serif',
   661→   661→            color: 'rgba(255,255,255,0.9)',
   662→   662→            textShadow: '0 1px 4px rgba(0,0,0,0.9), 0 0 8px rgba(0,0,0,0.6)',
   663→   663→            whiteSpace: 'nowrap',
   664→   664→            letterSpacing: '1px',
   665→   665→          }}>
   666→   666→            {territory.name}
   667→   667→          </div>
   668→   668→          {territory.units.length > 0 && dominant && (
   669→   669→            <div style={{
   670→   670→              marginTop: '2px',
   671→   671→              display: 'inline-flex',
   672→   672→              alignItems: 'center',
   673→   673→              gap: '2px',
   674→   674→              background: 'rgba(0,0,0,0.6)',
   675→   675→              borderRadius: '8px',
   676→   676→              padding: '1px 5px',
   677→   677→              border: `1px solid ${owner ? owner.color + '66' : 'rgba(139,115,85,0.3)'}`,
   678→   678→            }}>
   679→   679→              <span style={{ fontSize: '9px' }}>{UNIT_TYPES[dominant].icon}</span>
   680→   680→              <span style={{
   681→   681→                fontSize: '8px',
   682→   682→                fontWeight: 'bold',
   683→   683→                color: owner ? owner.colorLight : '#ccc',
   684→   684→                textShadow: '0 1px 2px rgba(0,0,0,0.8)',
   685→   685→              }}>
   686→   686→                {territory.units.length}
   687→   687→              </span>
   688→   688→            </div>
   689→   689→          )}
   690→   690→        </div>
   691→   691→      </Html>
   692→   692→    </group>
   693→   693→  );
   694→   694→}

   699→   699→function ConnectionLines() {
   700→   700→  const territories = useGameStore(s => s.territories);
   701→   701→
   702→   702→  const lines = useMemo(() => {
   703→   703→    const result: { start: [number, number, number]; end: [number, number, number] }[] = [];
   704→   704→    const drawn = new Set<string>();
   705→   705→    for (const t of Object.values(territories)) {
   706→   706→      const vis = TERRITORY_VISUAL[t.id];
   707→   707→      const h = (vis?.height || 0.2) + 0.1;
   708→   708→      for (const adjId of t.adjacentTo) {
   709→   709→        const key = [t.id, adjId].sort().join('-');
   710→   710→        if (!drawn.has(key)) {
   711→   711→          drawn.add(key);
   712→   712→          const adj = territories[adjId];
   713→   713→          if (adj) {
   714→   714→            const [x1, z1] = svgTo3D(t.labelX, t.labelY);
   715→   715→            const [x2, z2] = svgTo3D(adj.labelX, adj.labelY);
   716→   716→            const adjVis = TERRITORY_VISUAL[adjId];
   717→   717→            const h2 = (adjVis?.height || 0.2) + 0.1;
   718→   718→            const avgH = (h + h2) / 2 + 0.08;
   719→   719→            result.push({ start: [x1, avgH, z1], end: [x2, avgH, z2] });
   720→   720→          }
   721→   721→        }
   722→   722→      }
   723→   723→    }
   724→   724→    return result;
   725→   725→  }, [territories]);
   726→   726→
   727→   727→  return (
   728→   728→    <group>
   729→   729→      {lines.map((line, i) => (
   730→   730→        <Line
   731→   731→          key={`conn-${i}`}
   732→   732→          points={[line.start, line.end]}
   733→   733→          color="#8B735544"
   734→   734→          lineWidth={1}
   735→   735→          dashed
   736→   736→          dashSize={0.08}
   737→   737→          gapSize={0.08}
   738→   738→        />
   739→   739→      ))}
   740→   740→    </group>
   741→   741→  );
   742→   742→}

   845→   845→      pos[i * 3 + 1] = Math.random() * 10 + 0.5;
   846→   846→      pos[i * 3 + 2] = (Math.random() - 0.5) * 32;
   847→   847→    }

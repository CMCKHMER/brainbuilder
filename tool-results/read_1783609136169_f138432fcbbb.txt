    51→function parseSVGPath(pathStr: string): [number, number][] {
    52→  const points: [number, number][] = [];
    53→  const parts = pathStr.trim().split(/\s+/);
    54→  let i = 0;
    55→  while (i < parts.length) {
    56→    const cmd = parts[i];
    57→    if (cmd === 'M' || cmd === 'L') {
    58→      const x = parseFloat(parts[i + 1]);
    59→      const y = parseFloat(parts[i + 2]);
    60→      if (!isNaN(x) && !isNaN(y)) {
    61→        points.push([x, y]);
    62→      }
    63→      i += 3;
    64→    } else if (cmd === 'Z') {
    65→      break;
    66→    } else {
    67→      i++;
    68→    }
    69→  }
    70→  return points;
    71→}
    72→
    73→function svgTo3D(sx: number, sy: number): [number, number] {
    74→  return [(sx - CENTER_X) * SCALE, -(sy - CENTER_Y) * SCALE];
    75→}
    76→
    77→function createTerritoryGeometry(pathStr: string, height: number): THREE.ExtrudeGeometry {
    78→  const points = parseSVGPath(pathStr);
    79→  if (points.length < 3) {
    80→    return new THREE.ExtrudeGeometry(new THREE.Shape(), { depth: 0.1 });
    81→  }
    82→  const shape = new THREE.Shape();
    83→  const [x0, z0] = svgTo3D(points[0][0], points[0][1]);
    84→  shape.moveTo(x0, z0);
    85→  for (let i = 1; i < points.length; i++) {
    86→    const [x, z] = svgTo3D(points[i][0], points[i][1]);
    87→    shape.lineTo(x, z);
    88→  }
    89→  shape.closePath();
    90→
    91→  const geo = new THREE.ExtrudeGeometry(shape, {
    92→    steps: 1,
    93→    depth: height,
    94→    bevelEnabled: true,
    95→    bevelThickness: 0.015,
    96→    bevelSize: 0.015,
    97→    bevelOffset: 0,
    98→    bevelSegments: 2,
    99→  });
   100→  geo.rotateX(-Math.PI / 2);
   101→  geo.translate(0, 0.05, 0);
   102→  return geo;
   103→}

   399→function IceStructure({ position }: { position: [number, number, number] }) {

   401→  useFrame((state) => {
   402→    if (iceRef.current) {
   403→      iceRef.current.children.forEach((child, i) => {
   404→        child.position.y = (i === 0 ? 0.2 : i === 1 ? 0.1 : 0.04) + Math.sin(state.clock.elapsedTime * 1.5 + i) * 0.008;
   405→      });
   406→    }
   407→  });
   408→  return (
   409→    <group position={position} ref={iceRef}>
   410→      {/* Ice spire 1 */}
   411→      <mesh position={[0, 0.2, 0]}>

   447→
   448→interface TerritoryMeshProps {
   449→  territoryId: string;
   450→  path: string;
   451→  labelX: number;
   452→  labelY: number;
   453→  onHover: (id: string | null) => void;
   454→  onClick: (id: string) => void;
   455→}
   456→
   457→function TerritoryMesh({ territoryId, path, labelX, labelY, onHover, onClick }: TerritoryMeshProps) {
   458→  const territories = useGameStore(s => s.territories);
   459→  const selectedTerritory = useGameStore(s => s.selectedTerritory);
   460→  const targetTerritory = useGameStore(s => s.targetTerritory);
   461→  const phase = useGameStore(s => s.phase);
   462→  const players = useGameStore(s => s.players);
   463→  const reinforcementsLeft = useGameStore(s => s.reinforcementsLeft);
   464→  const deployUnitType = useGameStore(s => s.deployUnitType);
   465→  const currentPlayerIndex = useGameStore(s => s.currentPlayerIndex);
   466→  const selectTerritory = useGameStore(s => s.selectTerritory);
   467→  const deployArmy = useGameStore(s => s.deployArmy);
   468→
   469→  const territory = territories[territoryId];
   470→  if (!territory) return null;
   471→
   472→  const visual = TERRITORY_VISUAL[territoryId] || { structure: 'tower' as StructureType, height: 0.2, structureColor: '#8B7355', terrainColor: '#5A5A3A', terrainSideColor: '#3A3A2A' };
   473→  const playerColorMap = useMemo(() => {
   474→    const map: Record<string, { color: string; colorLight: string }> = {};
   475→    players.forEach(p => { map[p.id] = { color: p.color, colorLight: p.colorLight }; });
   476→    return map;
   477→  }, [players]);
   478→
   479→  const owner = territory.ownerId ? playerColorMap[territory.ownerId] : null;
   480→  const currentPlayerId = phase !== 'setup' && phase !== 'gameover' ? players[currentPlayerIndex]?.id : null;
   481→
   482→  const isSelected = territoryId === selectedTerritory;
   483→  const isTarget = territoryId === targetTerritory;
   484→
   485→  const geometry = useMemo(() => createTerritoryGeometry(path, visual.height), [path, visual.height]);
   486→  const edgesGeometry = useMemo(() => new THREE.EdgesGeometry(geometry, 25), [geometry]);
   487→
   488→  const [cx, cz] = svgTo3D(labelX, labelY);
   489→  const structY = 0.05 + visual.height;
   490→  const dominant = getDominantUnit(territory.units);
   491→
   492→  const isSelectable = useMemo(() => {
   493→    if (phase === 'deploy') {
   494→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   495→      return territory.ownerId === currentPlayerId && reinforcementsLeft >= cost;
   496→    }
   497→    if (phase === 'attack') {
   498→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   499→      if (territoryId === selectedTerritory) return true;
   500→      if (territory.ownerId !== currentPlayerId) return territory.adjacentTo.includes(selectedTerritory);
   501→      return territory.ownerId === currentPlayerId && territory.units.length > 1;
   502→    }
   503→    if (phase === 'fortify') {
   504→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   505→      if (territoryId === selectedTerritory) return true;
   506→      if (territory.ownerId === currentPlayerId && selectedTerritory !== null) return territory.adjacentTo.includes(selectedTerritory);
   507→      return false;
   508→    }
   509→    return false;
   510→  }, [phase, territory, selectedTerritory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex]);
   511→
   512→  const topColor = owner ? owner.colorLight : visual.terrainColor;
   513→  const sideColor = owner ? owner.color + 'AA' : visual.terrainSideColor;
   514→
   515→  const emissiveColor = isSelected ? '#FFD700' : isTarget ? '#FF0000' : '#000000';
   516→
   517→  const meshRef = useRef<THREE.Mesh>(null);
   518→  const edgesRef = useRef<THREE.LineSegments>(null);
   519→
   520→  useFrame((state) => {
   521→    if (meshRef.current) {
   522→      const mat = meshRef.current.material as THREE.MeshStandardMaterial;
   523→      if (isSelected) {
   524→        mat.emissiveIntensity = 0.3 + Math.sin(state.clock.elapsedTime * 3) * 0.15;
   525→      } else if (isTarget) {
   526→        mat.emissiveIntensity = 0.4 + Math.sin(state.clock.elapsedTime * 4) * 0.2;
   527→      } else {
   528→        mat.emissiveIntensity = 0;
   529→      }
   530→    }
   531→    if (edgesRef.current) {
   532→      const mat = edgesRef.current.material as THREE.LineBasicMaterial;
   533→      if (isSelected) {
   534→        mat.color.set('#FFD700');
   535→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 3) * 0.3;
   536→      } else if (isTarget) {
   537→        mat.color.set('#FF0000');
   538→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 4) * 0.3;
   539→      } else {
   540→        mat.color.set('#3D2B1F');
   541→        mat.opacity = 0.8;
   542→      }
   543→    }
   544→  });
   545→
   546→  const handleClick = useCallback((e: ThreeEvent<MouseEvent>) => {
   547→    e.stopPropagation();
   548→    if (phase === 'deploy') {
   549→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   550→      if (territory.ownerId === currentPlayerId && reinforcementsLeft >= cost) {
   551→        deployArmy(territoryId, deployUnitType);
   552→      }
   553→    } else if (phase === 'attack' || phase === 'fortify') {
   554→      selectTerritory(territoryId);
   555→    }
   556→  }, [phase, territory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex, selectTerritory, deployArmy, territoryId]);
   557→
   558→  const handlePointerOver = useCallback((e: ThreeEvent<PointerEvent>) => {
   559→    e.stopPropagation();
   560→    if (isSelectable) {
   561→      document.body.style.cursor = 'pointer';
   562→    }
   563→    onHover(territoryId);
   564→  }, [isSelectable, territoryId, onHover]);
   565→
   566→  const handlePointerOut = useCallback((e: ThreeEvent<PointerEvent>) => {
   567→    e.stopPropagation();
   568→    document.body.style.cursor = 'auto';
   569→    onHover(null);
   570→  }, [onHover]);
   571→
   572→  return (
   573→    <group>
   574→      {/* Territory extruded shape */}
   575→      <mesh
   576→        ref={meshRef}
   577→        geometry={geometry}
   578→        onClick={handleClick}
   579→        onPointerOver={handlePointerOver}
   580→        onPointerOut={handlePointerOut}
   581→      >
   582→        <meshStandardMaterial
   583→          color={topColor}
   584→          roughness={0.75}
   585→          metalness={0.05}
   586→          emissive={emissiveColor}
   587→          emissiveIntensity={0}
   588→          transparent={!isSelectable}
   589→          opacity={isSelectable ? 1 : (phase === 'attack' || phase === 'fortify') ? 0.6 : 0.85}
   590→        />
   591→      </mesh>
   592→
   593→      {/* Territory border edges */}
   594→      <lineSegments ref={edgesRef} geometry={edgesGeometry}>
   595→        <lineBasicMaterial color="#3D2B1F" transparent opacity={0.8} />
   596→      </lineSegments>
   597→
   598→      {/* 3D Structure */}
   599→      <StructureRenderer type={visual.structure} position={[cx, structY, cz]} color={visual.structureColor} />
   600→
   601→      {/* Territory label and unit badge */}
   602→      <Html position={[cx, structY + (visual.structure === 'mountain' ? 0.75 : visual.structure === 'tower' ? 0.8 : visual.structure === 'fortress' ? 0.45 : 0.35), cz]} center>
   603→        <div style={{ pointerEvents: 'none', userSelect: 'none', textAlign: 'center' }}>
   604→          <div style={{
   605→            fontSize: '9px',
   606→            fontFamily: 'var(--font-cinzel), serif',
   607→            color: 'rgba(255,255,255,0.9)',
   608→            textShadow: '0 1px 4px rgba(0,0,0,0.9), 0 0 8px rgba(0,0,0,0.6)',
   609→            whiteSpace: 'nowrap',
   610→            letterSpacing: '1px',
   611→          }}>
   612→            {territory.name}
   613→          </div>
   614→          {territory.units.length > 0 && dominant && (
   615→            <div style={{
   616→              marginTop: '2px',
   617→              display: 'inline-flex',
   618→              alignItems: 'center',
   619→              gap: '2px',
   620→              background: 'rgba(0,0,0,0.6)',
   621→              borderRadius: '8px',
   622→              padding: '1px 5px',
   623→              border: `1px solid ${owner ? owner.color + '66' : 'rgba(139,115,85,0.3)'}`,
   624→            }}>
   625→              <span style={{ fontSize: '9px' }}>{UNIT_TYPES[dominant].icon}</span>
   626→              <span style={{
   627→                fontSize: '8px',
   628→                fontWeight: 'bold',
   629→                color: owner ? owner.colorLight : '#ccc',
   630→                textShadow: '0 1px 2px rgba(0,0,0,0.8)',
   631→              }}>
   632→                {territory.units.length}
   633→              </span>
   634→            </div>
   635→          )}
   636→        </div>
   637→      </Html>
   638→    </group>
   639→  );
   640→}

   642→// ========================
   643→// Connection Lines
   644→// ========================
   645→function ConnectionLines() {
   646→  const territories = useGameStore(s => s.territories);
   647→
   648→  const lines = useMemo(() => {
   649→    const result: { start: [number, number, number]; end: [number, number, number] }[] = [];
   650→    const drawn = new Set<string>();
   651→    for (const t of Object.values(territories)) {
   652→      const vis = TERRITORY_VISUAL[t.id];
   653→      const h = (vis?.height || 0.2) + 0.1;
   654→      for (const adjId of t.adjacentTo) {
   655→        const key = [t.id, adjId].sort().join('-');
   656→        if (!drawn.has(key)) {
   657→          drawn.add(key);
   658→          const adj = territories[adjId];
   659→          if (adj) {
   660→            const [x1, z1] = svgTo3D(t.labelX, t.labelY);
   661→            const [x2, z2] = svgTo3D(adj.labelX, adj.labelY);
   662→            const adjVis = TERRITORY_VISUAL[adjId];
   663→            const h2 = (adjVis?.height || 0.2) + 0.1;
   664→            const avgH = (h + h2) / 2 + 0.08;
   665→            result.push({ start: [x1, avgH, z1], end: [x2, avgH, z2] });
   666→          }
   667→        }
   668→      }
   669→    }
   670→    return result;
   671→  }, [territories]);
   672→
   673→  return (
   674→    <group>
   675→      {lines.map((line, i) => (
   676→        <Line
   677→          key={`conn-${i}`}
   678→          points={[line.start, line.end]}
   679→          color="#8B735544"
   680→          lineWidth={1}
   681→          dashed
   682→          dashSize={0.08}
   683→          gapSize={0.08}
   684→        />
   685→      ))}
   686→    </group>
   687→  );
   688→}

   693→function FloatingPlatform() {

   721→        <meshStandardMaterial color="#2A1A0E" roughness={1} />

   726→        <meshStandardMaterial color="#1A0E06" roughness={1} />
   727→      </mesh>
   728→      {/* Bottom point */}
   729→      <mesh position={[0, -3.2, 0]}>
   730→        <coneGeometry args={[2.5, 1.2, 64]} />
   731→        <meshStandardMaterial color="#150A04" roughness={1} />
   732→      </mesh>
   733→      {/* Rocky protrusions on the bottom */}
   734→      {[0, 1, 2, 3, 4, 5].map((i) => {
   735→        const angle = (i / 6) * Math.PI * 2;
   736→        const r = 6.5 + Math.sin(i * 2.3) * 1.2;
   737→        const y = -1.5 - Math.random() * 1.5;
   738→        return (
   739→          <mesh key={i} position={[Math.cos(angle) * r, y, Math.sin(angle) * r]} rotation={[Math.random(), Math.random(), Math.random()]}>
   740→            <dodecahedronGeometry args={[0.3 + Math.random() * 0.4, 0]} />
   741→            <meshStandardMaterial color="#2A1A0E" roughness={1} />
   742→          </mesh>
   743→        );
   744→      })}
   745→    </group>
   746→  );
   747→}

   774→// ========================
   775→// Magical Particles
   776→// ========================
   777→function MagicalParticles() {
   778→  const pointsRef = useRef<THREE.Points>(null);
   779→  const count = 150;
   780→
   781→  const [positions] = useState(() => {
   782→    const pos = new Float32Array(count * 3);
   783→    for (let i = 0; i < count; i++) {
   784→      pos[i * 3] = (Math.random() - 0.5) * 16;
   785→      pos[i * 3 + 1] = Math.random() * 5 + 0.5;
   786→      pos[i * 3 + 2] = (Math.random() - 0.5) * 12;
   787→    }
   788→    return pos;
   789→  });
   790→
   791→  const [sizes] = useState(() => {
   792→    const s = new Float32Array(count);
   793→    for (let i = 0; i < count; i++) {
   794→      s[i] = Math.random() * 0.04 + 0.01;
   795→    }
   796→    return s;
   797→  });
   798→
   799→  useFrame((state) => {
   800→    if (pointsRef.current) {
   801→      pointsRef.current.rotation.y = state.clock.elapsedTime * 0.02;
   802→      const posAttr = pointsRef.current.geometry.attributes.position as THREE.BufferAttribute;
   803→      const arr = posAttr.array as Float32Array;
   804→      for (let i = 0; i < count; i++) {
   805→        arr[i * 3 + 1] += Math.sin(state.clock.elapsedTime + i) * 0.0005;
   806→      }
   807→      posAttr.needsUpdate = true;
   808→    }
   809→  });
   810→
   811→  return (
   812→    <points ref={pointsRef}>
   813→      <bufferGeometry>
   814→        <bufferAttribute
   815→          attach="attributes-position"
   816→          array={positions}
   817→          count={count}
   818→          itemSize={3}
   819→        />
   820→      </bufferGeometry>
   821→      <pointsMaterial
   822→        size={0.04}
   823→        color="#D4A017"
   824→        transparent
   825→        opacity={0.5}
   826→        sizeAttenuation
   827→        blending={THREE.AdditiveBlending}
   828→        depthWrite={false}
   829→      />
   830→    </points>
   831→  );
   832→}

   837→function SceneLighting() {
   838→  const dirLightRef = useRef<THREE.DirectionalLight>(null);
   839→
   840→  useFrame((state) => {
   841→    if (dirLightRef.current) {
   842→      const t = state.clock.elapsedTime * 0.1;
   843→      dirLightRef.current.position.x = Math.sin(t) * 3;
   844→      dirLightRef.current.position.z = Math.cos(t) * 3;
   845→    }
   846→  });
   847→
   848→  return (
   849→    <>
   850→      {/* Ambient light - cool moonlight */}
   851→      <ambientLight intensity={0.25} color="#4466AA" />
   852→      {/* Hemisphere light - sky/ground */}
   853→      <hemisphereLight args={['#2A3A5A', '#3A2A1A', 0.3]} />
   854→      {/* Main directional - slowly orbiting moonlight */}
   855→      <directionalLight
   856→        ref={dirLightRef}
   857→        position={[3, 8, 3]}
   858→        intensity={0.6}
   859→        color="#C0C8E0"
   860→        castShadow={false}
   861→      />
   862→      {/* Warm accent light - from below the platform */}
   863→      <pointLight position={[0, -2, 0]} intensity={0.4} color="#FF8844" distance={12} decay={2} />

   865→      <pointLight position={[-4, 4, -2]} intensity={0.15} color="#6688CC" distance={15} decay={2} />

   867→    </>
   868→  );

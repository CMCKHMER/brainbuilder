    77→function parseSVGPath(pathStr: string): [number, number][] {
    78→  const points: [number, number][] = [];
    79→  const parts = pathStr.trim().split(/\s+/);
    80→  let i = 0;
    81→  while (i < parts.length) {
    82→    const cmd = parts[i];
    83→    if (cmd === 'M' || cmd === 'L') {
    84→      const x = parseFloat(parts[i + 1]);
    85→      const y = parseFloat(parts[i + 2]);
    86→      if (!isNaN(x) && !isNaN(y)) {
    87→        points.push([x, y]);
    88→      }
    89→      i += 3;
    90→    } else if (cmd === 'Z') {
    91→      break;
    92→    } else {
    93→      i++;
    94→    }
    95→  }
    96→  return points;
    97→}

    99→function svgTo3D(sx: number, sy: number): [number, number] {
   100→  return [(sx - CENTER_X) * SCALE, -(sy - CENTER_Y) * SCALE];

   103→// ========================
   104→// Simple noise for terrain
   105→// ========================
   106→function simpleNoise(x: number, z: number): number {
   107→  const dot = x * 12.9898 + z * 78.233;
   108→  return (Math.sin(dot) * 43758.5453) % 1;
   109→}
   110→
   111→function smoothNoise(x: number, z: number): number {
   112→  const ix = Math.floor(x);
   113→  const iz = Math.floor(z);
   114→  const fx = x - ix;
   115→  const fz = z - iz;
   116→  const a = simpleNoise(ix, iz);
   117→  const b = simpleNoise(ix + 1, iz);
   118→  const c = simpleNoise(ix, iz + 1);
   119→  const d = simpleNoise(ix + 1, iz + 1);
   120→  const ux = fx * fx * (3 - 2 * fx);
   121→  const uz = fz * fz * (3 - 2 * fz);
   122→  return a + (b - a) * ux + (c - a) * uz + (a - b - c + d) * ux * uz;
   123→}
   124→
   125→function fbmNoise(x: number, z: number, octaves: number = 4): number {
   126→  let val = 0;
   127→  let amp = 1;
   128→  let freq = 1;
   129→  let max = 0;
   130→  for (let i = 0; i < octaves; i++) {
   131→    val += smoothNoise(x * freq, z * freq) * amp;
   132→    max += amp;
   133→    amp *= 0.5;
   134→    freq *= 2;
   135→  }
   136→  return val / max;
   137→}

   202→function sampleGradient(palette: RegionPalette, t: number): [number, number, number] {
   203→  t = Math.max(0, Math.min(1, t));
   204→  if (t < 0.33) {
   205→    const f = t / 0.33;
   206→    return [
   207→      palette.lowColor[0] + (palette.midColor[0] - palette.lowColor[0]) * f,
   208→      palette.lowColor[1] + (palette.midColor[1] - palette.lowColor[1]) * f,
   209→      palette.lowColor[2] + (palette.midColor[2] - palette.lowColor[2]) * f,
   210→    ];
   211→  } else if (t < 0.66) {
   212→    const f = (t - 0.33) / 0.33;
   213→    return [
   214→      palette.midColor[0] + (palette.highColor[0] - palette.midColor[0]) * f,
   215→      palette.midColor[1] + (palette.highColor[1] - palette.midColor[1]) * f,
   216→      palette.midColor[2] + (palette.highColor[2] - palette.midColor[2]) * f,
   217→    ];
   218→  } else {
   219→    const f = (t - 0.66) / 0.34;
   220→    return [
   221→      palette.highColor[0] + (palette.peakColor[0] - palette.highColor[0]) * f,
   222→      palette.highColor[1] + (palette.peakColor[1] - palette.highColor[1]) * f,
   223→      palette.highColor[2] + (palette.peakColor[2] - palette.highColor[2]) * f,
   224→    ];
   225→  }
   226→}

   228→function createTerritoryGeometry(
   229→  pathStr: string,
   230→  baseHeight: number,
   231→  territoryId: string,
   232→  region: string,
   233→): THREE.BufferGeometry {
   234→  const points = parseSVGPath(pathStr);
   235→  if (points.length < 3) {
   236→    return new THREE.ExtrudeGeometry(new THREE.Shape(), { depth: 0.1 });
   237→  }
   238→  const shape = new THREE.Shape();
   239→  const [x0, z0] = svgTo3D(points[0][0], points[0][1]);
   240→  shape.moveTo(x0, z0);
   241→  for (let i = 1; i < points.length; i++) {
   242→    const [x, z] = svgTo3D(points[i][0], points[i][1]);
   243→    shape.lineTo(x, z);
   244→  }
   245→  shape.closePath();
   246→
   247→  const geo = new THREE.ExtrudeGeometry(shape, {
   248→    steps: 1,
   249→    depth: baseHeight,
   250→    bevelEnabled: true,
   251→    bevelThickness: 0.015,
   252→    bevelSize: 0.015,
   253→    bevelOffset: 0,
   254→    bevelSegments: 2,
   255→  });
   256→  geo.rotateX(-Math.PI / 2);
   257→  geo.translate(0, 0.05, 0);
   258→
   259→  // ---- Vertex displacement for hills ----
   260→  const posAttr = geo.attributes.position;
   261→  const posArr = posAttr.array as Float32Array;
   262→  const vertexCount = posAttr.count;
   263→
   264→  // Find the Y range to identify top face vs sides
   265→  let minY = Infinity, maxY = -Infinity;
   266→  for (let i = 0; i < vertexCount; i++) {
   267→    const y = posArr[i * 3 + 1];
   268→    if (y < minY) minY = y;
   269→    if (y > maxY) maxY = y;
   270→  }
   271→  const topThreshold = maxY - 0.06; // Top face vertices
   272→
   273→  // Compute centroid for distance-based height falloff
   274→  let cx = 0, cz = 0, count = 0;
   275→  for (let i = 0; i < vertexCount; i++) {
   276→    if (posArr[i * 3 + 1] >= topThreshold) {
   277→      cx += posArr[i * 3];
   278→      cz += posArr[i * 3 + 2];
   279→      count++;
   280→    }
   281→  }
   282→  cx /= (count || 1);
   283→  cz /= (count || 1);
   284→
   285→  // Compute max distance from centroid for normalization
   286→  let maxDist = 0;
   287→  for (let i = 0; i < vertexCount; i++) {
   288→    if (posArr[i * 3 + 1] >= topThreshold) {
   289→      const dx = posArr[i * 3] - cx;
   290→      const dz = posArr[i * 3 + 2] - cz;
   291→      const d = Math.sqrt(dx * dx + dz * dz);
   292→      if (d > maxDist) maxDist = d;
   293→    }
   294→  }
   295→
   296→  // Hill height parameters per territory (seeded by territory name hash)
   297→  let seed = 0;
   298→  for (let c = 0; c < territoryId.length; c++) seed = seed * 31 + territoryId.charCodeAt(c);
   299→  const hillScale = 0.15 + (Math.abs(seed) % 100) / 100 * 0.2; // 0.15 - 0.35
   300→  const noiseScale = 3 + (Math.abs(seed * 7) % 100) / 100 * 4; // 3 - 7
   301→  const ridgeFactor = (Math.abs(seed * 13) % 100) / 100; // 0-1
   302→
   303→  // Displace top face vertices
   304→  const heights: number[] = new Array(vertexCount).fill(0);
   305→  for (let i = 0; i < vertexCount; i++) {
   306→    if (posArr[i * 3 + 1] >= topThreshold) {
   307→      const x = posArr[i * 3];
   308→      const z = posArr[i * 3 + 2];
   309→      const dx = x - cx;
   310→      const dz = z - cz;
   311→      const dist = Math.sqrt(dx * dx + dz * dz);
   312→      const distNorm = maxDist > 0 ? dist / maxDist : 0;
   313→
   314→      // Distance falloff: higher near center, lower at edges
   315→      const falloff = Math.max(0, 1 - distNorm * distNorm);
   316→
   317→      // Multi-octave noise for natural terrain
   318→      const n1 = fbmNoise(x * noiseScale + seed, z * noiseScale + seed * 0.7, 4);
   319→      const n2 = fbmNoise(x * noiseScale * 0.5 + seed * 2, z * noiseScale * 0.5 + seed, 3);
   320→
   321→      // Ridge noise: creates mountain ridges
   322→      const ridge = 1 - Math.abs(n1 * 2 - 1);
   323→      const heightVal = (n1 * 0.5 + n2 * 0.3 + ridge * ridgeFactor * 0.4) * hillScale * falloff;
   324→
   325→      posArr[i * 3 + 1] += heightVal;
   326→      heights[i] = heightVal;
   327→    }
   328→  }
   329→  posAttr.needsUpdate = true;
   330→  geo.computeVertexNormals();
   331→
   332→  // ---- Vertex colors for gradient shading ----
   333→  const palette = getRegionPalette(region);
   334→  const colors = new Float32Array(vertexCount * 3);
   335→
   336→  // Find the max height for normalization
   337→  let maxH = 0;
   338→  for (let i = 0; i < vertexCount; i++) {
   339→    if (heights[i] > maxH) maxH = heights[i];
   340→  }
   341→
   342→  for (let i = 0; i < vertexCount; i++) {
   343→    const y = posArr[i * 3 + 1];
   344→    const isTop = y >= topThreshold - 0.01;
   345→
   346→    if (isTop && maxH > 0) {
   347→      const t = heights[i] / maxH;
   348→      const [r, g, b] = sampleGradient(palette, t);
   349→      colors[i * 3] = r;
   350→      colors[i * 3 + 1] = g;
   351→      colors[i * 3 + 2] = b;
   352→    } else {
   353→      // Side faces: darken version of the low color
   354→      colors[i * 3] = palette.lowColor[0] * 0.6;
   355→      colors[i * 3 + 1] = palette.lowColor[1] * 0.6;
   356→      colors[i * 3 + 2] = palette.lowColor[2] * 0.6;
   357→    }
   358→  }
   359→
   360→  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   361→  return geo;
   362→}

   749→function TerritoryMesh({ territoryId, path, labelX, labelY, onHover, onClick }: TerritoryMeshProps) {
   750→  const territories = useGameStore(s => s.territories);
   751→  const selectedTerritory = useGameStore(s => s.selectedTerritory);
   752→  const targetTerritory = useGameStore(s => s.targetTerritory);
   753→  const phase = useGameStore(s => s.phase);
   754→  const players = useGameStore(s => s.players);
   755→  const reinforcementsLeft = useGameStore(s => s.reinforcementsLeft);
   756→  const deployUnitType = useGameStore(s => s.deployUnitType);
   757→  const currentPlayerIndex = useGameStore(s => s.currentPlayerIndex);
   758→  const selectTerritory = useGameStore(s => s.selectTerritory);
   759→  const deployArmy = useGameStore(s => s.deployArmy);
   760→  const battleAnimation = useGameStore(s => s.battleAnimation);
   761→
   762→  const territory = territories[territoryId];
   763→  if (!territory) return null;
   764→
   765→  const visual = TERRITORY_VISUAL[territoryId] || { structure: 'tower' as StructureType, height: 0.2, structureColor: '#8B7355', terrainColor: '#5A5A3A', terrainSideColor: '#3A3A2A' };
   766→  const playerColorMap = useMemo(() => {
   767→    const map: Record<string, { color: string; colorLight: string }> = {};
   768→    players.forEach(p => { map[p.id] = { color: p.color, colorLight: p.colorLight }; });
   769→    return map;
   770→  }, [players]);
   771→
   772→  const owner = territory.ownerId ? playerColorMap[territory.ownerId] : null;
   773→  const currentPlayerId = phase !== 'setup' && phase !== 'gameover' ? players[currentPlayerIndex]?.id : null;
   774→
   775→  const isSelected = territoryId === selectedTerritory;
   776→  const isTarget = territoryId === targetTerritory;
   777→
   778→  const geometry = useMemo(() => {
   779→    const tDef = TERRITORY_DEFS.find(t => t.id === territoryId);
   780→    const region = tDef?.region || '';
   781→    return createTerritoryGeometry(path, visual.height, territoryId, region);
   782→  }, [path, visual.height, territoryId]);
   783→  const edgesGeometry = useMemo(() => new THREE.EdgesGeometry(geometry, 25), [geometry]);
   784→
   785→  const [cx, cz] = svgTo3D(labelX, labelY);
   786→  const structY = 0.05 + visual.height;
   787→  const dominant = getDominantUnit(territory.units);
   788→
   789→  const isSelectable = useMemo(() => {
   790→    if (phase === 'deploy') {
   791→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   792→      return territory.ownerId === currentPlayerId && reinforcementsLeft >= cost;
   793→    }
   794→    if (phase === 'attack') {
   795→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   796→      if (territoryId === selectedTerritory) return true;
   797→      if (territory.ownerId !== currentPlayerId) return territory.adjacentTo.includes(selectedTerritory);
   798→      return territory.ownerId === currentPlayerId && territory.units.length > 1;
   799→    }
   800→    if (phase === 'fortify') {
   801→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   802→      if (territoryId === selectedTerritory) return true;
   803→      if (territory.ownerId === currentPlayerId && selectedTerritory !== null) return territory.adjacentTo.includes(selectedTerritory);
   804→      return false;
   805→    }
   806→    return false;
   807→  }, [phase, territory, selectedTerritory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex]);
   808→
   809→  const topColor = owner ? owner.colorLight : visual.terrainColor;
   810→  const sideColor = owner ? owner.color + 'AA' : visual.terrainSideColor;
   811→
   812→  const emissiveColor = isSelected ? '#FFD700' : isTarget ? '#FF0000' : '#000000';
   813→
   814→  const meshRef = useRef<THREE.Mesh>(null);
   815→  const edgesRef = useRef<THREE.LineSegments>(null);
   816→
   817→  useFrame((state) => {
   818→    if (!meshRef.current) return;
   819→
   820→    const mat = meshRef.current.material as THREE.MeshStandardMaterial;
   821→    const edgeMat = edgesRef.current?.material as THREE.LineBasicMaterial | undefined;
   822→
   823→    // Battle animation effects
   824→    const isBattleTarget = battleAnimation?.toTerritory === territoryId;
   825→    const isBattleSource = battleAnimation?.fromTerritory === territoryId;
   826→    const battleAge = battleAnimation ? (Date.now() - battleAnimation.timestamp) / 1000 : 999;
   827→
   828→    if (isBattleTarget && battleAge < 2) {
   829→      // Target: flash red, pulsing emissive
   830→      const flash = Math.max(0, 1 - battleAge / 2);
   831→      const pulse = Math.sin(battleAge * 15) * 0.5 + 0.5;
   832→      mat.emissive.set(battleAnimation.conquered ? '#FF6600' : '#FF0000');
   833→      mat.emissiveIntensity = flash * (0.5 + pulse * 0.5);
   834→      // Shake effect
   835→      const shake = Math.max(0, 1 - battleAge / 1.5) * 0.08;
   836→      meshRef.current.position.y = Math.sin(battleAge * 30) * shake;
   837→      meshRef.current.position.x = Math.cos(battleAge * 25) * shake * 0.5;
   838→      if (edgeMat) {
   839→        edgeMat.color.set(battleAnimation.conquered ? '#FF8800' : '#FF3333');
   840→        edgeMat.opacity = 0.6 + flash * 0.4;
   841→      }
   842→    } else if (isBattleSource && battleAge < 1.5) {
   843→      // Source: brief gold flash
   844→      const flash = Math.max(0, 1 - battleAge / 1.5);
   845→      mat.emissive.set('#FFD700');
   846→      mat.emissiveIntensity = flash * 0.3;
   847→      meshRef.current.position.y = 0;
   848→      meshRef.current.position.x = 0;
   849→      if (edgeMat) {
   850→        edgeMat.color.set('#FFD700');
   851→        edgeMat.opacity = 0.6 + flash * 0.3;
   852→      }
   853→    } else {
   854→      // Normal selection/target logic
   855→      meshRef.current.position.y = 0;
   856→      meshRef.current.position.x = 0;
   857→
   858→      if (isSelected) {
   859→        mat.emissive.set('#FFD700');
   860→        mat.emissiveIntensity = 0.3 + Math.sin(state.clock.elapsedTime * 3) * 0.15;
   861→      } else if (isTarget) {
   862→        mat.emissive.set('#FF0000');
   863→        mat.emissiveIntensity = 0.4 + Math.sin(state.clock.elapsedTime * 4) * 0.2;
   864→      } else {
   865→        mat.emissiveIntensity = 0;
   866→      }
   867→
   868→      if (edgeMat) {
   869→        if (isSelected) {
   870→          edgeMat.color.set('#FFD700');
   871→          edgeMat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 3) * 0.3;
   872→        } else if (isTarget) {
   873→          edgeMat.color.set('#FF0000');
   874→          edgeMat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 4) * 0.3;
   875→        } else {
   876→          edgeMat.color.set('#3D2B1F');
   877→          edgeMat.opacity = 0.8;
   878→        }
   879→      }
   880→    }
   881→  });
   882→
   883→  const handleClick = useCallback((e: ThreeEvent<MouseEvent>) => {
   884→    e.stopPropagation();
   885→    if (phase === 'deploy') {
   886→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   887→      if (territory.ownerId === currentPlayerId && reinforcementsLeft >= cost) {
   888→        deployArmy(territoryId, deployUnitType);

  1346→      vels[i * 3] = Math.cos(angle) * speed;

  1348→      vels[i * 3 + 2] = Math.sin(angle) * speed;
  1349→    }
  1350→    return vels;
  1351→  });
  1352→  const startTimeRef = useRef(0);
  1353→
  1354→  useEffect(() => {

  1356→      const targetDef = TERRITORY_DEFS_LOCAL.find(t => t.id === battleAnimation.toTerritory);
  1357→      if (!targetDef) return;

  1360→
  1361→      // Initialize particle positions at the target location
  1362→      const pos = new Float32Array(40 * 3);
  1363→      for (let i = 0; i < 40; i++) {
  1364→        pos[i * 3] = cx + (Math.random() - 0.5) * 0.3;
  1365→        pos[i * 3 + 1] = height;
  1366→        pos[i * 3 + 2] = cz + (Math.random() - 0.5) * 0.3;
  1367→      }
  1368→      setPositions(pos);
  1369→      setActive(true);
  1370→      startTimeRef.current = Date.now();
  1371→    }
  1372→  }, [battleAnimation, territories, TERRITORY_DEFS_LOCAL]);
  1373→
  1374→  useFrame(() => {
  1375→    if (!active || !pointsRef.current || !positions) return;

  1380→    }
  1381→
  1382→    const posArr = pointsRef.current.geometry.attributes.position.array as Float32Array;
  1383→    for (let i = 0; i < 40; i++) {
  1384→      const t = age;
  1385→      posArr[i * 3] = posArr[i * 3] + velocities[i * 3] * 0.008;
  1386→      posArr[i * 3 + 1] = posArr[i * 3 + 1] + velocities[i * 3 + 1] * 0.008 - t * t * 2 * 0.008; // gravity
  1387→      posArr[i * 3 + 2] = posArr[i * 3 + 2] + velocities[i * 3 + 2] * 0.008;
  1388→    }
  1389→    pointsRef.current.geometry.attributes.position.needsUpdate = true;
  1390→
  1391→    // Fade out
  1392→    const mat = pointsRef.current.material as THREE.PointsMaterial;
  1393→    mat.opacity = Math.max(0, 1 - age / 1.5);
  1394→    mat.size = 0.12 * (1 + age * 0.5);
  1395→  });
  1396→
  1397→  if (!active || !positions) return null;
  1398→
  1399→  return (
  1400→    <points ref={pointsRef}>
  1401→      <bufferGeometry>
  1402→        <bufferAttribute
  1403→          attach="attributes-position"
  1404→          args={[positions, 3]}
  1405→        />
  1406→      </bufferGeometry>
  1407→      <pointsMaterial
  1408→        size={0.12}
  1409→        color={battleAnimation?.conquered ? '#FF8800' : '#FF3333'}
  1410→        transparent
  1411→        opacity={1}
  1412→        sizeAttenuation
  1413→        blending={THREE.AdditiveBlending}
  1414→        depthWrite={false}
  1415→      />
  1416→    </points>
  1417→  );
  1418→}

  1645→interface GameMap3DProps {

  1648→
  1649→export default function GameMap3D({ onTerritoryHover = () => {} }: GameMap3DProps) {
  1650→  const [webGL, setWebGL] = useState<boolean | null>(null);
  1651→  const handleHover = useCallback((id: string | null) => {
  1652→    onTerritoryHover(id);
  1653→  }, [onTerritoryHover]);
  1654→
  1655→  useEffect(() => {
  1656→    setWebGL(isWebGLSupported());
  1657→  }, []);
  1658→
  1659→  // Still detecting
  1660→  if (webGL === null) return <LoadingScreen />;
  1661→
  1662→  // WebGL not available — show 2D fallback
  1663→  if (!webGL) return <Fallback2DMap onTerritoryHover={handleHover} />;
  1664→
  1665→  return (
  1666→    <Suspense fallback={<LoadingScreen />}>
  1667→      <Canvas
  1668→        camera={{ position: [0, 13, 12], fov: 62, near: 0.1, far: 200 }}
  1669→        gl={{ antialias: true, alpha: false, powerPreference: 'high-performance' }}
  1670→        style={{ background: '#060810' }}
  1671→        onPointerMissed={() => {
  1672→          useGameStore.getState().clearSelection();
  1673→        }}
  1674→        onError={() => {
  1675→          // If Canvas still fails after detection, show fallback
  1676→          setWebGL(false);
  1677→        }}
  1678→      >
  1679→        <WorldScene onTerritoryHover={handleHover} />
  1680→      </Canvas>
  1681→    </Suspense>
  1682→  );
  1683→}

   959→   959→export function getRivalDialogue(characterClass: string, context: RivalDialogue['context'], turnNumber: number): { text: string } | null {
   960→   960→  const cls = characterClass.toLowerCase();
   961→   961→  const eligible = RIVAL_DIALOGUES.filter(d => {
   962→   962→    if (d.speakerClass !== cls) return false;
   963→   963→    if (d.context !== context) return false;
   964→   964→    if (turnNumber < d.minTurn) return false;
   965→   965→    return true;
   966→   966→  });
   967→   967→  if (eligible.length === 0) return null;
   968→   968→  const dialogue = eligible[Math.floor(Math.random() * eligible.length)];
   969→   969→  const line = dialogue.lines[Math.floor(Math.random() * dialogue.lines.length)];
   970→   970→  return { text: line };
   971→   971→}

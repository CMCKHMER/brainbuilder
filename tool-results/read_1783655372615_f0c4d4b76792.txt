    54→    54→function parseSVGPath(pathStr: string): [number, number][] {
    55→    55→  const points: [number, number][] = [];
    56→    56→  const parts = pathStr.trim().split(/\s+/);
    57→    57→  let i = 0;
    58→    58→  while (i < parts.length) {
    59→    59→    const cmd = parts[i];
    60→    60→    if (cmd === 'M' || cmd === 'L') {
    61→    61→      const x = parseFloat(parts[i + 1]);
    62→    62→      const y = parseFloat(parts[i + 2]);
    63→    63→      if (!isNaN(x) && !isNaN(y)) {
    64→    64→        points.push([x, y]);
    65→    65→      }
    66→    66→      i += 3;
    67→    67→    } else if (cmd === 'Z') {
    68→    68→      break;
    69→    69→    } else {
    70→    70→      i++;
    71→    71→    }
    72→    72→  }
    73→    73→  return points;
    74→    74→}
    75→    75→
    76→    76→function svgTo3D(sx: number, sy: number): [number, number] {
    77→    77→  return [(sx - CENTER_X) * SCALE, -(sy - CENTER_Y) * SCALE];
    78→    78→}

    80→    80→function createTerritoryGeometry(pathStr: string, height: number): THREE.ExtrudeGeometry {
    81→    81→  const points = parseSVGPath(pathStr);
    82→    82→  if (points.length < 3) {
    83→    83→    return new THREE.ExtrudeGeometry(new THREE.Shape(), { depth: 0.1 });
    84→    84→  }
    85→    85→  const shape = new THREE.Shape();
    86→    86→  const [x0, z0] = svgTo3D(points[0][0], points[0][1]);
    87→    87→  shape.moveTo(x0, z0);
    88→    88→  for (let i = 1; i < points.length; i++) {
    89→    89→    const [x, z] = svgTo3D(points[i][0], points[i][1]);
    90→    90→    shape.lineTo(x, z);
    91→    91→  }
    92→    92→  shape.closePath();
    93→    93→
    94→    94→  const geo = new THREE.ExtrudeGeometry(shape, {
    95→    95→    steps: 1,
    96→    96→    depth: height,
    97→    97→    bevelEnabled: true,
    98→    98→    bevelThickness: 0.015,
    99→    99→    bevelSize: 0.015,
   100→   100→    bevelOffset: 0,
   101→   101→    bevelSegments: 2,
   102→   102→  });
   103→   103→  geo.rotateX(-Math.PI / 2);
   104→   104→  geo.translate(0, 0.05, 0);
   105→   105→  return geo;
   106→   106→}

   484→   484→interface TerritoryMeshProps {

   493→   493→function TerritoryMesh({ territoryId, path, labelX, labelY, onHover, onClick }: TerritoryMeshProps) {
   494→   494→  const territories = useGameStore(s => s.territories);
   495→   495→  const selectedTerritory = useGameStore(s => s.selectedTerritory);
   496→   496→  const targetTerritory = useGameStore(s => s.targetTerritory);
   497→   497→  const phase = useGameStore(s => s.phase);
   498→   498→  const players = useGameStore(s => s.players);
   499→   499→  const reinforcementsLeft = useGameStore(s => s.reinforcementsLeft);
   500→   500→  const deployUnitType = useGameStore(s => s.deployUnitType);
   501→   501→  const currentPlayerIndex = useGameStore(s => s.currentPlayerIndex);
   502→   502→  const selectTerritory = useGameStore(s => s.selectTerritory);
   503→   503→  const deployArmy = useGameStore(s => s.deployArmy);
   504→   504→
   505→   505→  const territory = territories[territoryId];
   506→   506→  if (!territory) return null;
   507→   507→
   508→   508→  const visual = TERRITORY_VISUAL[territoryId] || { structure: 'tower' as StructureType, height: 0.2, structureColor: '#8B7355', terrainColor: '#5A5A3A', terrainSideColor: '#3A3A2A' };
   509→   509→  const playerColorMap = useMemo(() => {
   510→   510→    const map: Record<string, { color: string; colorLight: string }> = {};
   511→   511→    players.forEach(p => { map[p.id] = { color: p.color, colorLight: p.colorLight }; });
   512→   512→    return map;
   513→   513→  }, [players]);
   514→   514→
   515→   515→  const owner = territory.ownerId ? playerColorMap[territory.ownerId] : null;
   516→   516→  const currentPlayerId = phase !== 'setup' && phase !== 'gameover' ? players[currentPlayerIndex]?.id : null;
   517→   517→
   518→   518→  const isSelected = territoryId === selectedTerritory;
   519→   519→  const isTarget = territoryId === targetTerritory;
   520→   520→
   521→   521→  const geometry = useMemo(() => createTerritoryGeometry(path, visual.height), [path, visual.height]);
   522→   522→  const edgesGeometry = useMemo(() => new THREE.EdgesGeometry(geometry, 25), [geometry]);
   523→   523→
   524→   524→  const [cx, cz] = svgTo3D(labelX, labelY);
   525→   525→  const structY = 0.05 + visual.height;
   526→   526→  const dominant = getDominantUnit(territory.units);
   527→   527→
   528→   528→  const isSelectable = useMemo(() => {
   529→   529→    if (phase === 'deploy') {
   530→   530→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   531→   531→      return territory.ownerId === currentPlayerId && reinforcementsLeft >= cost;
   532→   532→    }
   533→   533→    if (phase === 'attack') {
   534→   534→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   535→   535→      if (territoryId === selectedTerritory) return true;
   536→   536→      if (territory.ownerId !== currentPlayerId) return territory.adjacentTo.includes(selectedTerritory);
   537→   537→      return territory.ownerId === currentPlayerId && territory.units.length > 1;
   538→   538→    }
   539→   539→    if (phase === 'fortify') {
   540→   540→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   541→   541→      if (territoryId === selectedTerritory) return true;
   542→   542→      if (territory.ownerId === currentPlayerId && selectedTerritory !== null) return territory.adjacentTo.includes(selectedTerritory);
   543→   543→      return false;
   544→   544→    }
   545→   545→    return false;
   546→   546→  }, [phase, territory, selectedTerritory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex]);
   547→   547→
   548→   548→  const topColor = owner ? owner.colorLight : visual.terrainColor;
   549→   549→  const sideColor = owner ? owner.color + 'AA' : visual.terrainSideColor;
   550→   550→
   551→   551→  const emissiveColor = isSelected ? '#FFD700' : isTarget ? '#FF0000' : '#000000';
   552→   552→
   553→   553→  const meshRef = useRef<THREE.Mesh>(null);
   554→   554→  const edgesRef = useRef<THREE.LineSegments>(null);
   555→   555→
   556→   556→  useFrame((state) => {
   557→   557→    if (meshRef.current) {
   558→   558→      const mat = meshRef.current.material as THREE.MeshStandardMaterial;
   559→   559→      if (isSelected) {
   560→   560→        mat.emissiveIntensity = 0.3 + Math.sin(state.clock.elapsedTime * 3) * 0.15;
   561→   561→      } else if (isTarget) {
   562→   562→        mat.emissiveIntensity = 0.4 + Math.sin(state.clock.elapsedTime * 4) * 0.2;
   563→   563→      } else {
   564→   564→        mat.emissiveIntensity = 0;
   565→   565→      }
   566→   566→    }
   567→   567→    if (edgesRef.current) {
   568→   568→      const mat = edgesRef.current.material as THREE.LineBasicMaterial;
   569→   569→      if (isSelected) {
   570→   570→        mat.color.set('#FFD700');
   571→   571→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 3) * 0.3;
   572→   572→      } else if (isTarget) {
   573→   573→        mat.color.set('#FF0000');
   574→   574→        mat.opacity = 0.6 + Math.sin(state.clock.elapsedTime * 4) * 0.3;
   575→   575→      } else {
   576→   576→        mat.color.set('#3D2B1F');
   577→   577→        mat.opacity = 0.8;
   578→   578→      }
   579→   579→    }
   580→   580→  });
   581→   581→
   582→   582→  const handleClick = useCallback((e: ThreeEvent<MouseEvent>) => {
   583→   583→    e.stopPropagation();
   584→   584→    if (phase === 'deploy') {
   585→   585→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   586→   586→      if (territory.ownerId === currentPlayerId && reinforcementsLeft >= cost) {
   587→   587→        deployArmy(territoryId, deployUnitType);
   588→   588→      }
   589→   589→    } else if (phase === 'attack' || phase === 'fortify') {
   590→   590→      selectTerritory(territoryId);
   591→   591→    }
   592→   592→  }, [phase, territory, currentPlayerId, reinforcementsLeft, deployUnitType, players, currentPlayerIndex, selectTerritory, deployArmy, territoryId]);
   593→   593→
   594→   594→  const handlePointerOver = useCallback((e: ThreeEvent<PointerEvent>) => {
   595→   595→    e.stopPropagation();
   596→   596→    if (isSelectable) {
   597→   597→      document.body.style.cursor = 'pointer';
   598→   598→    }
   599→   599→    onHover(territoryId);
   600→   600→  }, [isSelectable, territoryId, onHover]);
   601→   601→
   602→   602→  const handlePointerOut = useCallback((e: ThreeEvent<PointerEvent>) => {
   603→   603→    e.stopPropagation();
   604→   604→    document.body.style.cursor = 'auto';
   605→   605→    onHover(null);
   606→   606→  }, [onHover]);
   607→   607→
   608→   608→  return (
   609→   609→    <group>
   610→   610→      {/* Territory extruded shape */}
   611→   611→      <mesh
   612→   612→        ref={meshRef}
   613→   613→        geometry={geometry}
   614→   614→        onClick={handleClick}
   615→   615→        onPointerOver={handlePointerOver}
   616→   616→        onPointerOut={handlePointerOut}
   617→   617→      >
   618→   618→        <meshStandardMaterial
   619→   619→          color={topColor}
   620→   620→          roughness={0.75}
   621→   621→          metalness={0.05}
   622→   622→          emissive={emissiveColor}
   623→   623→          emissiveIntensity={0}
   624→   624→          transparent={!isSelectable}
   625→   625→          opacity={isSelectable ? 1 : (phase === 'attack' || phase === 'fortify') ? 0.6 : 0.85}
   626→   626→        />
   627→   627→      </mesh>
   628→   628→
   629→   629→      {/* Territory border edges */}
   630→   630→      <lineSegments ref={edgesRef} geometry={edgesGeometry}>
   631→   631→        <lineBasicMaterial color="#3D2B1F" transparent opacity={0.8} />
   632→   632→      </lineSegments>
   633→   633→
   634→   634→      {/* 3D Structure - positioned so base sits on top of territory */}
   635→   635→      <StructureRenderer type={visual.structure} position={[cx, structY, cz]} color={visual.structureColor} />
   636→   636→
   637→   637→      {/* Territory label and unit badge */}
   638→   638→      <Html position={[cx, structY + (visual.structure === 'mountain' ? 2.8 : visual.structure === 'tower' ? 3.0 : visual.structure === 'fortress' ? 1.8 : visual.structure === 'forest' ? 1.2 : 1.5), cz]} center>
   639→   639→        <div style={{ pointerEvents: 'none', userSelect: 'none', textAlign: 'center' }}>
   640→   640→          <div style={{
   641→   641→            fontSize: '9px',
   642→   642→            fontFamily: 'var(--font-cinzel), serif',
   643→   643→            color: 'rgba(255,255,255,0.9)',
   644→   644→            textShadow: '0 1px 4px rgba(0,0,0,0.9), 0 0 8px rgba(0,0,0,0.6)',
   645→   645→            whiteSpace: 'nowrap',
   646→   646→            letterSpacing: '1px',
   647→   647→          }}>
   648→   648→            {territory.name}
   649→   649→          </div>
   650→   650→          {territory.units.length > 0 && dominant && (
   651→   651→            <div style={{
   652→   652→              marginTop: '2px',
   653→   653→              display: 'inline-flex',
   654→   654→              alignItems: 'center',
   655→   655→              gap: '2px',
   656→   656→              background: 'rgba(0,0,0,0.6)',
   657→   657→              borderRadius: '8px',
   658→   658→              padding: '1px 5px',
   659→   659→              border: `1px solid ${owner ? owner.color + '66' : 'rgba(139,115,85,0.3)'}`,
   660→   660→            }}>
   661→   661→              <span style={{ fontSize: '9px' }}>{UNIT_TYPES[dominant].icon}</span>
   662→   662→              <span style={{
   663→   663→                fontSize: '8px',
   664→   664→                fontWeight: 'bold',
   665→   665→                color: owner ? owner.colorLight : '#ccc',
   666→   666→                textShadow: '0 1px 2px rgba(0,0,0,0.8)',
   667→   667→              }}>
   668→   668→                {territory.units.length}
   669→   669→              </span>
   670→   670→            </div>
   671→   671→          )}
   672→   672→        </div>
   673→   673→      </Html>
   674→   674→    </group>
   675→   675→  );
   676→   676→}

   681→   681→function ConnectionLines() {
   682→   682→  const territories = useGameStore(s => s.territories);
   683→   683→
   684→   684→  const lines = useMemo(() => {
   685→   685→    const result: { start: [number, number, number]; end: [number, number, number] }[] = [];
   686→   686→    const drawn = new Set<string>();
   687→   687→    for (const t of Object.values(territories)) {
   688→   688→      const vis = TERRITORY_VISUAL[t.id];
   689→   689→      const h = (vis?.height || 0.2) + 0.1;
   690→   690→      for (const adjId of t.adjacentTo) {
   691→   691→        const key = [t.id, adjId].sort().join('-');
   692→   692→        if (!drawn.has(key)) {
   693→   693→          drawn.add(key);
   694→   694→          const adj = territories[adjId];
   695→   695→          if (adj) {
   696→   696→            const [x1, z1] = svgTo3D(t.labelX, t.labelY);
   697→   697→            const [x2, z2] = svgTo3D(adj.labelX, adj.labelY);
   698→   698→            const adjVis = TERRITORY_VISUAL[adjId];
   699→   699→            const h2 = (adjVis?.height || 0.2) + 0.1;
   700→   700→            const avgH = (h + h2) / 2 + 0.08;
   701→   701→            result.push({ start: [x1, avgH, z1], end: [x2, avgH, z2] });
   702→   702→          }
   703→   703→        }
   704→   704→      }
   705→   705→    }
   706→   706→    return result;
   707→   707→  }, [territories]);
   708→   708→
   709→   709→  return (
   710→   710→    <group>
   711→   711→      {lines.map((line, i) => (
   712→   712→        <Line
   713→   713→          key={`conn-${i}`}
   714→   714→          points={[line.start, line.end]}
   715→   715→          color="#8B735544"
   716→   716→          lineWidth={1}
   717→   717→          dashed
   718→   718→          dashSize={0.08}
   719→   719→          gapSize={0.08}
   720→   720→        />
   721→   721→      ))}
   722→   722→    </group>
   723→   723→  );
   724→   724→}

   819→   819→function MagicalParticles() {
   820→   820→  const pointsRef = useRef<THREE.Points>(null);
   821→   821→  const count = 150;
   822→   822→
   823→   823→  const [positions] = useState(() => {
   824→   824→    const pos = new Float32Array(count * 3);
   825→   825→    for (let i = 0; i < count; i++) {
   826→   826→      pos[i * 3] = (Math.random() - 0.5) * 22;
   827→   827→      pos[i * 3 + 1] = Math.random() * 6 + 0.5;
   828→   828→      pos[i * 3 + 2] = (Math.random() - 0.5) * 16;
   829→   829→    }
   830→   830→    return pos;
   831→   831→  });
   832→   832→
   833→   833→  const [sizes] = useState(() => {
   834→   834→    const s = new Float32Array(count);
   835→   835→    for (let i = 0; i < count; i++) {
   836→   836→      s[i] = Math.random() * 0.04 + 0.01;
   837→   837→    }
   838→   838→    return s;
   839→   839→  });
   840→   840→
   841→   841→  useFrame((state) => {
   842→   842→    if (pointsRef.current) {
   843→   843→      pointsRef.current.rotation.y = state.clock.elapsedTime * 0.02;
   844→   844→      const posAttr = pointsRef.current.geometry.attributes.position as THREE.BufferAttribute;
   845→   845→      const arr = posAttr.array as Float32Array;
   846→   846→      for (let i = 0; i < count; i++) {
   847→   847→        arr[i * 3 + 1] += Math.sin(state.clock.elapsedTime + i) * 0.0005;
   848→   848→      }
   849→   849→      posAttr.needsUpdate = true;
   850→   850→    }
   851→   851→  });
   852→   852→
   853→   853→  return (
   854→   854→    <points ref={pointsRef}>
   855→   855→      <bufferGeometry>
   856→   856→        <bufferAttribute
   857→   857→          attach="attributes-position"
   858→   858→          array={positions}
   859→   859→          count={count}
   860→   860→          itemSize={3}
   861→   861→        />
   862→   862→      </bufferGeometry>
   863→   863→      <pointsMaterial
   864→   864→        size={0.04}
   865→   865→        color="#D4A017"
   866→   866→        transparent
   867→   867→        opacity={0.5}
   868→   868→        sizeAttenuation
   869→   869→        blending={THREE.AdditiveBlending}
   870→   870→        depthWrite={false}
   871→   871→      />
   872→   872→    </points>
   873→   873→  );
   874→   874→}

   141→function randomlyAssignTerritories(playerCount: number, players: Player[]): Record<string, TerritoryState> {
   142→  const territories: Record<string, TerritoryState> = {};
   143→  const territoryIds = TERRITORIES.map(t => t.id);
   144→  // Shuffle
   145→  for (let i = territoryIds.length - 1; i > 0; i--) {
   146→    const j = Math.floor(Math.random() * (i + 1));
   147→    [territoryIds[i], territoryIds[j]] = [territoryIds[j], territoryIds[i]];
   148→  }
   149→
   150→  // Default unit types to distribute
   151→  const defaultUnits: UnitTypeId[] = ['spearman', 'swordsman', 'swordsman', 'archer', 'cavalry', 'shield_bearer', 'mage', 'assassin'];
   152→
   153→  for (let i = 0; i < territoryIds.length; i++) {
   154→    const def = TERRITORIES.find(t => t.id === territoryIds[i])!;
   155→    const playerIndex = i % playerCount;
   156→    const unitType = defaultUnits[i % defaultUnits.length];
   157→    territories[territoryIds[i]] = {
   158→      ...def,
   159→      ownerId: players[playerIndex].id,
   160→      units: [unitType], // Start with 1 unit each
   161→    };
   162→  }
   163→
   164→  // Give each player bonus units to reach minimum 2-3 units per territory on average
   165→  const territoriesPerPlayer = Math.floor(16 / playerCount);
   166→  for (const player of players) {
   167→    const playerTerritories = Object.values(territories).filter(t => t.ownerId === player.id);
   168→    const bonusUnits = Math.max(0, Math.ceil(territoriesPerPlayer / 2));
   169→    for (let i = 0; i < bonusUnits; i++) {
   170→      const randomTerritory = playerTerritories[Math.floor(Math.random() * playerTerritories.length)];
   171→      const randomUnit = defaultUnits[Math.floor(Math.random() * defaultUnits.length)];
   172→      territories[randomTerritory.id].units.push(randomUnit);
   173→    }
   174→  }
   175→
   176→  return territories;
   177→}

   278→
   279→  deployArmy: (territoryId, unitType) => {

   282→

   285→

   289→

   291→    newTerritories[territoryId] = {

   310→  selectTerritory: (territoryId) => {
   311→    const state = get();
   312→    const territory = state.territories[territoryId];
   313→    if (!territory) return;
   314→    const currentPlayer = state.players[state.currentPlayerIndex];
   315→
   316→    if (state.phase === 'attack') {
   317→      if (state.selectedTerritory === null) {
   318→        if (territory.ownerId === currentPlayer.id && territory.units.length > 1) {
   319→          set({
   320→            selectedTerritory: territoryId,
   321→            targetTerritory: null,
   322→            attackerDiceCount: 0,
   323→            battleResult: null,
   324→          });
   325→        }

   327→        set({ selectedTerritory: null, targetTerritory: null, attackerDiceCount: 0, battleResult: null });
   328→      } else if (territory.ownerId !== currentPlayer.id) {
   329→        const selectedTerritory = state.territories[state.selectedTerritory];
   330→        if (selectedTerritory.adjacentTo.includes(territoryId)) {
   331→          set({
   332→            targetTerritory: territoryId,
   333→            defenderDiceCount: getMaxDefenderDice(territory.units.length),
   334→          });
   335→        }
   336→      } else if (territory.ownerId === currentPlayer.id && territory.units.length > 1) {
   337→        set({
   338→          selectedTerritory: territoryId,
   339→          targetTerritory: null,
   340→          attackerDiceCount: 0,
   341→          battleResult: null,
   342→        });
   343→      }
   344→    }
   345→
   346→    if (state.phase === 'fortify') {
   347→      if (state.selectedTerritory === null) {
   348→        if (territory.ownerId === currentPlayer.id && territory.units.length > 1) {
   349→          set({
   350→            selectedTerritory: territoryId,
   351→            targetTerritory: null,
   352→            fortifyArmies: territory.units.length - 1,
   353→          });
   354→        }
   355→      } else if (state.selectedTerritory === territoryId) {
   356→        set({ selectedTerritory: null, targetTerritory: null, fortifyArmies: 0 });
   357→      } else if (territory.ownerId === currentPlayer.id) {
   358→        const selectedTerritory = state.territories[state.selectedTerritory];
   359→        if (selectedTerritory.adjacentTo.includes(territoryId)) {
   360→          set({
   361→            targetTerritory: territoryId,
   362→            fortifyArmies: Math.min(state.fortifyArmies, selectedTerritory.units.length - 1),
   363→          });
   364→        }
   365→      }
   366→    }
   367→  },
   368→
   369→  clearSelection: () => {
   370→    set({ selectedTerritory: null, targetTerritory: null, attackerDiceCount: 0, defenderDiceCount: 0, battleResult: null, fortifyArmies: 0 });
   371→  },
   372→
   373→  setAttackerDiceCount: (count) => set({ attackerDiceCount: count }),
   374→
   375→  executeAttack: () => {
   376→    const state = get();
   377→    if (!state.selectedTerritory || !state.targetTerritory || state.attackerDiceCount === 0) return;
   378→
   379→    const fromTerritory = state.territories[state.selectedTerritory];
   380→    const toTerritory = state.territories[state.targetTerritory];
   381→    if (!fromTerritory || !toTerritory) return;
   382→
   383→    const attacker = state.players.find(p => p.id === fromTerritory.ownerId)!;
   384→    const defender = state.players.find(p => p.id === toTerritory.ownerId)!;
   385→
   386→    // Get tactic modifiers
   387→    const modifiers = state.getActiveTacticEffect();
   388→    const activeTactic = state.selectedTactic;
   389→
   390→    const result = resolveBattle(
   391→      state.attackerDiceCount,
   392→      state.defenderDiceCount,
   393→      fromTerritory.units,
   394→      toTerritory.units,
   395→      attacker.characterClass,
   396→      defender.characterClass,
   397→      modifiers || undefined,
   398→      activeTactic,
   399→    );
   400→
   401→    const newTerritories = { ...state.territories };
   402→    const newPlayers = state.players.map(p => ({ ...p, territories: [...p.territories] }));
   403→
   404→    // Apply losses using unit system
   405→    newTerritories[state.selectedTerritory] = {
   406→      ...fromTerritory,
   407→      units: removeUnits(fromTerritory.units, result.attackerLosses),
   408→    };
   409→    newTerritories[state.targetTerritory] = {
   410→      ...toTerritory,
   411→      units: removeUnits(toTerritory.units, result.defenderLosses),
   412→    };
   413→
   414→    let newLog = [...state.battleLog];
   415→    let conquered = false;
   416→
   417→    // Build advantage label
   418→    const advantageLabel = result.typeAdvantage === 'strong' ? ' ⬆ADVANTAGE' : result.typeAdvantage === 'weak' ? ' ⬇DISADVANTAGE' : '';
   419→    const atkTypeName = result.attackerType ? UNIT_TYPES[result.attackerType].name : 'Army';
   420→    const defTypeName = result.defenderType ? UNIT_TYPES[result.defenderType].name : 'Army';
   421→
   422→    if (newTerritories[state.targetTerritory].units.length === 0) {
   423→      conquered = true;
   424→      const movingUnits = Math.min(state.attackerDiceCount, newTerritories[state.selectedTerritory].units.length);
   425→
   426→      const oldOwner = newTerritories[state.targetTerritory].ownerId;
   427→      const newOwner = newTerritories[state.selectedTerritory].ownerId;
   428→
   429→      // Move top units to conquered territory
   430→      const remaining = newTerritories[state.selectedTerritory].units;
   431→      const movedUnits = remaining.slice(0, movingUnits);
   432→      const stayedUnits = remaining.slice(movingUnits);
   433→
   434→      newTerritories[state.targetTerritory] = {
   435→        ...newTerritories[state.targetTerritory],
   436→        ownerId: newOwner,
   437→        units: movedUnits,
   438→      };
   439→      newTerritories[state.selectedTerritory] = {
   440→        ...newTerritories[state.selectedTerritory],
   441→        units: stayedUnits,
   442→      };
   443→
   444→      if (oldOwner) {
   445→        const oldPlayer = newPlayers.find(p => p.id === oldOwner)!;
   446→        oldPlayer.territories = oldPlayer.territories.filter(id => id !== state.targetTerritory);
   447→        if (oldPlayer.territories.length === 0) {
   448→          oldPlayer.eliminated = true;
   449→          newLog.push(addLog(state, `💀 ${oldPlayer.name} has been eliminated!`, 'info'));
   450→
   451→          // Queue elimination story for AI players
   452→          if (oldPlayer.isAI) {
   453→            const elimClass = oldPlayer.characterClass.toLowerCase();
   454→            const elimConfig = PLAYER_CONFIGS.find(p => p.characterClass.toLowerCase() === elimClass) || PLAYER_CONFIGS[0];
   455→            const elimBeat = getEliminationBeat(oldPlayer.name, elimClass, oldPlayer.color);
   456→            get().queueStory(elimBeat);
   457→          }
   458→        }
   459→      }
   460→      const newOwnerPlayer = newPlayers.find(p => p.id === newOwner)!;
   461→      newOwnerPlayer.territories.push(state.targetTerritory);
   462→
   463→      newLog.push(addLog(state, `🏆 ${attacker.name} conquered ${toTerritory.name} from ${defender.name}!`, 'conquer'));
   464→
   465→      // Campaign story triggers on conquest
   466→      if (get().isCampaignMode) {
   467→        get().checkStoryTriggers(state.targetTerritory, fromTerritory.ownerId, toTerritory.ownerId);
   468→      }
   469→    } else {
   470→      newLog.push(addLog(state,
   471→        `⚔️ ${atkTypeName}${advantageLabel} → ${defTypeName}: ATK lost ${result.attackerLosses}, DEF lost ${result.defenderLosses}`,
   472→        'attack'
   473→      ));
   474→    }
   475→
   476→    const battleResult: BattleResult = {
   477→      ...result,
   478→      conquered,
   479→      fromTerritory: state.selectedTerritory,
   480→      toTerritory: state.targetTerritory,
   481→      tacticUsed: activeTactic,
   482→    };
   483→
   484→    // Check win condition
   485→    const winner = newPlayers.find(p => !p.eliminated && p.territories.length === 16) || null;
   486→
   487→    // Queue victory story if game ends
   488→    if (winner) {
   489→      const winClass = winner.characterClass.toLowerCase();
   490→      const winConfig = PLAYER_CONFIGS.find(p => p.characterClass.toLowerCase() === winClass) || PLAYER_CONFIGS[0];
   491→      const victoryBeat = getVictoryBeat(winner.name, winClass, winner.color, winner.colorLight, winConfig.image);
   492→      get().queueStory(victoryBeat);
   493→    }
   494→
   495→    const maxAttack = getMaxAttackerDice(newTerritories[state.selectedTerritory].units.length);
   496→    const maxDefend = getMaxDefenderDice(newTerritories[state.targetTerritory].units.length);
   497→
   498→    // Mark tactic as used if it's an attack-phase tactic and battle happened
   499→    let newActiveTactics = state.activeTactics;
   500→    if (activeTactic) {
   501→      const tactic = TACTICS[activeTactic];
   502→      if (tactic.phase === 'attack') {
   503→        newActiveTactics = newActiveTactics.map(t =>
   504→          t.tacticId === activeTactic ? { ...t, usedThisTurn: true } : t
   505→        );
   506→      }
   507→    }
   508→
   509→    set({
   510→      territories: newTerritories,
   511→      players: newPlayers,
   512→      battleResult,
   513→      battleLog: newLog,
   514→      attackerDiceCount: conquered ? 0 : Math.min(state.attackerDiceCount, maxAttack),
   515→      defenderDiceCount: conquered ? 0 : maxDefend,
   516→      targetTerritory: conquered ? null : state.targetTerritory,
   517→      selectedTerritory: conquered ? null : state.selectedTerritory,
   518→      winner: winner ? winner : null,
   519→      phase: winner ? 'gameover' : state.phase,
   520→      activeTactics: newActiveTactics,
   521→    });
   522→  },
   523→
   524→  setFortifyArmies: (count) => set({ fortifyArmies: count }),
   525→
   526→  executeFortify: (fromId, toId) => {
   527→    const state = get();
   528→    if (state.fortifyArmies <= 0) return;
   529→
   530→    const newTerritories = { ...state.territories };
   531→    const fromUnits = newTerritories[fromId].units;
   532→    const toUnits = newTerritories[toId].units;
   533→
   534→    // Move last N units (weakest) to destination
   535→    const moved = fromUnits.slice(-state.fortifyArmies);
   536→    const remaining = fromUnits.slice(0, fromUnits.length - state.fortifyArmies);
   537→
   538→    newTerritories[fromId] = {
   539→      ...newTerritories[fromId],
   540→      units: remaining,
   541→    };
   542→    newTerritories[toId] = {
   543→      ...newTerritories[toId],
   544→      units: [...toUnits, ...moved],
   545→    };
   546→
   547→    const fromName = newTerritories[fromId].name;
   548→    const toName = newTerritories[toId].name;
   549→    const newLog = [...state.battleLog, addLog(state, `🛡️ ${state.fortifyArmies} units moved: ${fromName} → ${toName}`, 'fortify')];
   550→
   551→    set({
   552→      territories: newTerritories,
   553→      battleLog: newLog,
   554→      selectedTerritory: null,
   555→      targetTerritory: null,
   556→      fortifyArmies: 0,
   557→    });

   691→  endTurn: () => {
   692→    const state = get();
   693→    const nextPlayerIndex = (state.currentPlayerIndex + 1) % state.players.length;
   694→    let actualNext = nextPlayerIndex;
   695→    let safety = 0;
   696→
   697→    while (state.players[actualNext].eliminated && safety < 10) {
   698→      actualNext = (actualNext + 1) % state.players.length;
   699→      safety++;
   700→    }
   701→
   702→    // If we looped back to the current player, it means only they remain
   703→    const newTurnNumber = actualNext <= state.currentPlayerIndex ? state.turnNumber + 1 : state.turnNumber;
   704→
   705→    const nextPlayer = state.players[actualNext];
   706→
   707→    // Check if rally cry is active for next player
   708→    const rallyCryActive = state.activeTactics.some(
   709→      t => t.tacticId === 'rally_cry' && t.playerId === nextPlayer.id && t.turnUsed === newTurnNumber && !t.usedThisTurn
   710→    );
   711→
   712→    const reinforcements = calculateReinforcements(
   713→      nextPlayer.territories.length,
   714→      nextPlayer.characterClass,
   715→      rallyCryActive,
   716→    );
   717→
   718→    // Mark rally cry as used if it was active
   719→    let newActiveTactics = state.activeTactics;
   720→    if (rallyCryActive) {
   721→      newActiveTactics = newActiveTactics.map(t =>
   722→        t.tacticId === 'rally_cry' && t.playerId === nextPlayer.id && t.turnUsed === newTurnNumber
   723→          ? { ...t, usedThisTurn: true }
   724→          : t
   725→      );
   726→    }
   727→
   728→    const newLog = [...state.battleLog, addLog(state, `--- ${nextPlayer.name}'s turn ---`, 'turn')];
   729→
   730→    set({
   731→      phase: 'deploy',
   732→      currentPlayerIndex: actualNext,
   733→      turnNumber: newTurnNumber,
   734→      reinforcementsLeft: reinforcements,
   735→      selectedTerritory: null,
   736→      targetTerritory: null,
   737→      attackerDiceCount: 0,
   738→      defenderDiceCount: 0,
   739→      battleResult: null,
   740→      battleLog: newLog,
   741→      fortifyArmies: 0,
   742→      selectedTactic: null,
   743→      activeTactics: newActiveTactics,
   744→      deployUnitType: 'swordsman',
   745→    });
   746→
   747→    // Roll for campaign event on new turn cycle (only for human player's turn)
   748→    if (newTurnNumber > state.turnNumber && !nextPlayer.isAI) {
   749→      get().rollCampaignEvent();
   750→
   751→      // Campaign: check chapter advancement and turn-based triggers
   752→      if (get().isCampaignMode) {
   753→        get().advanceChapter();
   754→      }

   828→  dismissStory: () => {
   829→    const state = get();

   837→      // Don't replay seen beats
   838→      if (seen.has(next.id)) {
   839→        // Skip this one and try the next
   840→        set({ storyBeat: queue.length > 0 ? queue.shift()! : null, storyQueue: queue, storySeen: seen });
   841→      } else {
   842→        set({ storyBeat: next, storyQueue: queue, storySeen: seen });
   843→      }
   844→    } else {
   845→      set({ storyBeat: null, storyQueue: queue, storySeen: seen });
   846→    }

   849→  rollCampaignEvent: () => {
   850→    const state = get();
   851→    if (state.phase === 'gameover' || state.storyBeat) return;
   852→
   853→    const eligible = CAMPAIGN_EVENTS.filter(e => {
   854→      if (state.turnNumber < e.minTurn) return false;
   855→      if (e.maxTurn && state.turnNumber > e.maxTurn) return false;
   856→      if (state.storySeen.has(e.id)) return false;
   857→      return Math.random() < e.triggerChance;
   858→    });
   859→
   860→    if (eligible.length > 0) {
   861→      const event = eligible[Math.floor(Math.random() * eligible.length)];
   862→      const seen = new Set(state.storySeen);
   863→      seen.add(event.id);
   864→      set({ currentEvent: event, storySeen: seen });

   894→  advanceChapter: () => {
   895→    const state = get();
   896→    if (!state.isCampaignMode || state.phase === 'gameover' || state.storyBeat) return;
   897→    if (state.storyQueue.length > 0) return; // Don't advance while story is queued
   898→
   899→    const progress = { ...state.campaignProgress };
   900→    const humanPlayer = state.players.find(p => !p.isAI) || state.players[0];
   901→    if (!humanPlayer) return;
   902→
   903→    const territoryCount = humanPlayer.territories.length;
   904→    const alivePlayers = state.players.filter(p => !p.eliminated).length;
   905→
   906→    let newChapter = progress.currentChapter;
   907→
   908→    // Chapter 2: First conquest happened
   909→    if (newChapter === 1 && progress.totalConquests >= 1) {
   910→      newChapter = 2;
   911→    }
   912→    // Chapter 3: A player has been eliminated OR turn 8+
   913→    else if (newChapter === 2 && (alivePlayers < state.players.length || state.turnNumber >= 8)) {
   914→      newChapter = 3;
   915→    }
   916→    // Chapter 4: Human has 8+ territories OR only 2 players remain
   917→    else if (newChapter === 3 && (territoryCount >= 8 || alivePlayers <= 2)) {
   918→      newChapter = 4;
   919→    }
   920→    // Chapter 5: Only 2 players remain
   921→    else if (newChapter === 4 && alivePlayers <= 2) {
   922→      newChapter = 5;
   923→    }
   924→
   925→    if (newChapter !== progress.currentChapter) {
   926→      progress.currentChapter = newChapter;
   927→      const chapter = CHAPTERS.find(c => c.number === newChapter);
   928→      if (chapter) {
   929→        const beat = getChapterTitleBeat(chapter);
   930→        get().queueStory(beat);
   931→      }
   932→      set({ campaignProgress: progress });
   933→    }
   934→  },
   935→
   936→  checkStoryTriggers: (conqueredTerritoryId, attackerId, defenderId) => {
   937→    const state = get();
   938→    if (!state.isCampaignMode || state.phase === 'gameover') return;
   939→
   940→    const progress = { ...state.campaignProgress };
   941→    const firedTriggers = new Set(progress.firedTriggers);
   942→    const rivalClashes = new Set(progress.rivalClashes);
   943→    const regionDominanceFired = new Set(progress.regionDominanceFired);
   944→
   945→    // Increment conquest counter
   946→    progress.totalConquests = (progress.totalConquests || 0) + 1;
   947→
   948→    if (!conqueredTerritoryId || !attackerId || !defenderId) {
   949→      set({ campaignProgress: progress });
   950→      return;
   951→    }
   952→
   953→    const attacker = state.players.find(p => p.id === attackerId);
   954→    const defender = state.players.find(p => p.id === defenderId);
   955→    if (!attacker || !defender) return;
   956→
   957→    const territory = state.territories[conqueredTerritoryId];
   958→    if (!territory) return;
   959→
   960→    // Get config for portrait lookups
   961→    const atkConfig = PLAYER_CONFIGS.find(p => p.characterClass.toLowerCase() === attacker.characterClass.toLowerCase()) || PLAYER_CONFIGS[0];
   962→    const defConfig = PLAYER_CONFIGS.find(p => p.characterClass.toLowerCase() === defender.characterClass.toLowerCase()) || PLAYER_CONFIGS[1];
   963→
   964→    // --- FIRST BLOOD ---
   965→    if (!progress.firstBloodFired) {
   966→      progress.firstBloodFired = true;
   967→      const beat = getFirstBloodBeat(
   968→        attacker.name, attacker.characterClass, attacker.color, attacker.colorLight, atkConfig.image,
   969→        defender.name, territory.name
   970→      );
   971→      get().queueStory(beat);
   972→    }
   973→
   974→    // --- KEY TERRITORY CAPTURE ---
   975→    const keyTerritoryTriggerId = `territory-${conqueredTerritoryId}`;
   976→    if (!firedTriggers.has(keyTerritoryTriggerId)) {
   977→      const captureBeat = getTerritoryCaptureBeat(
   978→        conqueredTerritoryId, territory.name, attacker.name, attacker.characterClass,
   979→        attacker.color, attacker.colorLight, atkConfig.image
   980→      );
   981→      if (captureBeat) {
   982→        firedTriggers.add(keyTerritoryTriggerId);
   983→        get().queueStory(captureBeat);
   984→      }
   985→    }
   986→
   987→    // --- RIVAL CLASH ---
   988→    const clashKey = [attackerId, defenderId].sort().join('-');
   989→    if (!rivalClashes.has(clashKey)) {
   990→      rivalClashes.add(clashKey);
   991→      const clashBeat = getRivalClashBeat(
   992→        attacker.name, attacker.characterClass, attacker.color, atkConfig.image,
   993→        defender.name, defender.characterClass, defender.color, defConfig.image
   994→      );
   995→      get().queueStory(clashBeat);
   996→    }
   997→
   998→    // --- REGION DOMINANCE ---
   999→    const region = territory.region;
  1000→    if (!regionDominanceFired.has(region)) {
  1001→      // Check if attacker now controls all territories in this region
  1002→      const regionTerritories = TERRITORIES.filter(t => t.region === region);
  1003→      const ownsAll = regionTerritories.every(t => {
  1004→        const tState = state.territories[t.id];
  1005→        return tState && tState.ownerId === attackerId;
  1006→      });
  1007→      if (ownsAll && regionTerritories.length >= 2) {
  1008→        regionDominanceFired.add(region);
  1009→        const lore = REGION_LORE[region] || '';
  1010→        const dominanceBeat = getRegionDominanceBeat(region, attacker.name, attacker.color, lore);
  1011→        get().queueStory(dominanceBeat);
  1012→      }
  1013→    }
  1014→
  1015→    // --- DESPERATE HOUR (check all human players) ---
  1016→    for (const player of state.players) {
  1017→      if (player.isAI || player.eliminated) continue;
  1018→      const pConfig = PLAYER_CONFIGS.find(p => p.characterClass.toLowerCase() === player.characterClass.toLowerCase()) || PLAYER_CONFIGS[0];
  1019→      const desperateId = `desperate-${player.id}`;
  1020→      if (!firedTriggers.has(desperateId) && player.territories.length <= 2 && state.turnNumber >= 3) {
  1021→        firedTriggers.add(desperateId);
  1022→        const desperateBeat = getDesperateHourBeat(
  1023→          player.name, player.characterClass, player.color, player.colorLight, pConfig.image,
  1024→          player.territories.length
  1025→        );
  1026→        get().queueStory(desperateBeat);
  1027→      }
  1028→    }
  1029→
  1030→    // --- DOMINANT FORCE ---
  1031→    for (const player of state.players) {
  1032→      if (player.eliminated) continue;
  1033→      const dominantId = `dominant-${player.id}`;
  1034→      if (!firedTriggers.has(dominantId) && player.territories.length >= 10 && state.turnNumber >= 5) {
  1035→        firedTriggers.add(dominantId);
  1036→        const dominantBeat = getDominantForceBeat(
  1037→          player.name, player.color, player.territories.length, state.players.filter(p => !p.eliminated).length
  1038→        );
  1039→        get().queueStory(dominantBeat);
  1040→      }
  1041→    }
  1042→
  1043→    progress.firedTriggers = firedTriggers;
  1044→    progress.rivalClashes = rivalClashes;
  1045→    progress.regionDominanceFired = regionDominanceFired;
  1046→    set({ campaignProgress: progress });
  1047→  },

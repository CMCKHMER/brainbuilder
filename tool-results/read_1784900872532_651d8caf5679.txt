    77→    77→function parseSVGPath(pathStr: string): [number, number][] {
    78→    78→  const points: [number, number][] = [];
    79→    79→  const parts = pathStr.trim().split(/\s+/);
    80→    80→  let i = 0;
    81→    81→  while (i < parts.length) {
    82→    82→    const cmd = parts[i];
    83→    83→    if (cmd === 'M' || cmd === 'L') {
    84→    84→      const x = parseFloat(parts[i + 1]);
    85→    85→      const y = parseFloat(parts[i + 2]);
    86→    86→      if (!isNaN(x) && !isNaN(y)) {
    87→    87→        points.push([x, y]);
    88→    88→      }
    89→    89→      i += 3;
    90→    90→    } else if (cmd === 'Z') {
    91→    91→      break;
    92→    92→    } else {
    93→    93→      i++;
    94→    94→    }
    95→    95→  }
    96→    96→  return points;
    97→    97→}

   104→   104→// Simple noise for terrain
   105→   105→// ========================
   106→   106→function simpleNoise(x: number, z: number): number {
   107→   107→  const dot = x * 12.9898 + z * 78.233;
   108→   108→  return (Math.sin(dot) * 43758.5453) % 1;
   109→   109→}
   110→   110→
   111→   111→function smoothNoise(x: number, z: number): number {
   112→   112→  const ix = Math.floor(x);
   113→   113→  const iz = Math.floor(z);
   114→   114→  const fx = x - ix;
   115→   115→  const fz = z - iz;
   116→   116→  const a = simpleNoise(ix, iz);
   117→   117→  const b = simpleNoise(ix + 1, iz);
   118→   118→  const c = simpleNoise(ix, iz + 1);
   119→   119→  const d = simpleNoise(ix + 1, iz + 1);
   120→   120→  const ux = fx * fx * (3 - 2 * fx);
   121→   121→  const uz = fz * fz * (3 - 2 * fz);
   122→   122→  return a + (b - a) * ux + (c - a) * uz + (a - b - c + d) * ux * uz;
   123→   123→}
   124→   124→
   125→   125→function fbmNoise(x: number, z: number, octaves: number = 4): number {
   126→   126→  let val = 0;
   127→   127→  let amp = 1;
   128→   128→  let freq = 1;
   129→   129→  let max = 0;
   130→   130→  for (let i = 0; i < octaves; i++) {
   131→   131→    val += smoothNoise(x * freq, z * freq) * amp;
   132→   132→    max += amp;
   133→   133→    amp *= 0.5;
   134→   134→    freq *= 2;
   135→   135→  }
   136→   136→  return val / max;
   137→   137→}

   202→   202→function sampleGradient(palette: RegionPalette, t: number): [number, number, number] {
   203→   203→  t = Math.max(0, Math.min(1, t));
   204→   204→  if (t < 0.33) {
   205→   205→    const f = t / 0.33;
   206→   206→    return [
   207→   207→      palette.lowColor[0] + (palette.midColor[0] - palette.lowColor[0]) * f,
   208→   208→      palette.lowColor[1] + (palette.midColor[1] - palette.lowColor[1]) * f,
   209→   209→      palette.lowColor[2] + (palette.midColor[2] - palette.lowColor[2]) * f,
   210→   210→    ];
   211→   211→  } else if (t < 0.66) {
   212→   212→    const f = (t - 0.33) / 0.33;
   213→   213→    return [
   214→   214→      palette.midColor[0] + (palette.highColor[0] - palette.midColor[0]) * f,
   215→   215→      palette.midColor[1] + (palette.highColor[1] - palette.midColor[1]) * f,
   216→   216→      palette.midColor[2] + (palette.highColor[2] - palette.midColor[2]) * f,
   217→   217→    ];
   218→   218→  } else {
   219→   219→    const f = (t - 0.66) / 0.34;
   220→   220→    return [
   221→   221→      palette.highColor[0] + (palette.peakColor[0] - palette.highColor[0]) * f,
   222→   222→      palette.highColor[1] + (palette.peakColor[1] - palette.highColor[1]) * f,
   223→   223→      palette.highColor[2] + (palette.peakColor[2] - palette.highColor[2]) * f,
   224→   224→    ];
   225→   225→  }
   226→   226→}

   228→   228→function createTerritoryGeometry(
   229→   229→  pathStr: string,
   230→   230→  baseHeight: number,
   231→   231→  territoryId: string,
   232→   232→  region: string,
   233→   233→): THREE.BufferGeometry {
   234→   234→  const points = parseSVGPath(pathStr);
   235→   235→  if (points.length < 3) {
   236→   236→    return new THREE.ExtrudeGeometry(new THREE.Shape(), { depth: 0.1 });
   237→   237→  }
   238→   238→  const shape = new THREE.Shape();
   239→   239→  const [x0, z0] = svgTo3D(points[0][0], points[0][1]);
   240→   240→  shape.moveTo(x0, z0);
   241→   241→  for (let i = 1; i < points.length; i++) {
   242→   242→    const [x, z] = svgTo3D(points[i][0], points[i][1]);
   243→   243→    shape.lineTo(x, z);
   244→   244→  }
   245→   245→  shape.closePath();
   246→   246→
   247→   247→  const geo = new THREE.ExtrudeGeometry(shape, {
   248→   248→    steps: 1,
   249→   249→    depth: baseHeight,
   250→   250→    bevelEnabled: true,
   251→   251→    bevelThickness: 0.015,
   252→   252→    bevelSize: 0.015,
   253→   253→    bevelOffset: 0,
   254→   254→    bevelSegments: 2,
   255→   255→  });
   256→   256→  geo.rotateX(-Math.PI / 2);
   257→   257→  geo.translate(0, 0.05, 0);
   258→   258→
   259→   259→  // ---- Vertex displacement for hills ----
   260→   260→  const posAttr = geo.attributes.position;
   261→   261→  const posArr = posAttr.array as Float32Array;
   262→   262→  const vertexCount = posAttr.count;
   263→   263→
   264→   264→  // Find the Y range to identify top face vs sides
   265→   265→  let minY = Infinity, maxY = -Infinity;
   266→   266→  for (let i = 0; i < vertexCount; i++) {
   267→   267→    const y = posArr[i * 3 + 1];
   268→   268→    if (y < minY) minY = y;
   269→   269→    if (y > maxY) maxY = y;
   270→   270→  }
   271→   271→  const topThreshold = maxY - 0.06; // Top face vertices
   272→   272→
   273→   273→  // Compute centroid for distance-based height falloff
   274→   274→  let cx = 0, cz = 0, count = 0;
   275→   275→  for (let i = 0; i < vertexCount; i++) {
   276→   276→    if (posArr[i * 3 + 1] >= topThreshold) {
   277→   277→      cx += posArr[i * 3];
   278→   278→      cz += posArr[i * 3 + 2];
   279→   279→      count++;
   280→   280→    }
   281→   281→  }
   282→   282→  cx /= (count || 1);
   283→   283→  cz /= (count || 1);
   284→   284→
   285→   285→  // Compute max distance from centroid for normalization
   286→   286→  let maxDist = 0;
   287→   287→  for (let i = 0; i < vertexCount; i++) {
   288→   288→    if (posArr[i * 3 + 1] >= topThreshold) {
   289→   289→      const dx = posArr[i * 3] - cx;
   290→   290→      const dz = posArr[i * 3 + 2] - cz;
   291→   291→      const d = Math.sqrt(dx * dx + dz * dz);
   292→   292→      if (d > maxDist) maxDist = d;
   293→   293→    }
   294→   294→  }
   295→   295→
   296→   296→  // Hill height parameters per territory (seeded by territory name hash)
   297→   297→  let seed = 0;
   298→   298→  for (let c = 0; c < territoryId.length; c++) seed = seed * 31 + territoryId.charCodeAt(c);
   299→   299→  const hillScale = 0.15 + (Math.abs(seed) % 100) / 100 * 0.2; // 0.15 - 0.35
   300→   300→  const noiseScale = 3 + (Math.abs(seed * 7) % 100) / 100 * 4; // 3 - 7
   301→   301→  const ridgeFactor = (Math.abs(seed * 13) % 100) / 100; // 0-1
   302→   302→
   303→   303→  // Displace top face vertices
   304→   304→  const heights: number[] = new Array(vertexCount).fill(0);
   305→   305→  for (let i = 0; i < vertexCount; i++) {
   306→   306→    if (posArr[i * 3 + 1] >= topThreshold) {
   307→   307→      const x = posArr[i * 3];
   308→   308→      const z = posArr[i * 3 + 2];
   309→   309→      const dx = x - cx;
   310→   310→      const dz = z - cz;
   311→   311→      const dist = Math.sqrt(dx * dx + dz * dz);
   312→   312→      const distNorm = maxDist > 0 ? dist / maxDist : 0;
   313→   313→
   314→   314→      // Distance falloff: higher near center, lower at edges
   315→   315→      const falloff = Math.max(0, 1 - distNorm * distNorm);
   316→   316→
   317→   317→      // Multi-octave noise for natural terrain
   318→   318→      const n1 = fbmNoise(x * noiseScale + seed, z * noiseScale + seed * 0.7, 4);
   319→   319→      const n2 = fbmNoise(x * noiseScale * 0.5 + seed * 2, z * noiseScale * 0.5 + seed, 3);
   320→   320→
   321→   321→      // Ridge noise: creates mountain ridges
   322→   322→      const ridge = 1 - Math.abs(n1 * 2 - 1);
   323→   323→      const heightVal = (n1 * 0.5 + n2 * 0.3 + ridge * ridgeFactor * 0.4) * hillScale * falloff;
   324→   324→
   325→   325→      posArr[i * 3 + 1] += heightVal;
   326→   326→      heights[i] = heightVal;
   327→   327→    }
   328→   328→  }
   329→   329→  posAttr.needsUpdate = true;
   330→   330→  geo.computeVertexNormals();
   331→   331→
   332→   332→  // ---- Vertex colors for gradient shading ----
   333→   333→  const palette = getRegionPalette(region);
   334→   334→  const colors = new Float32Array(vertexCount * 3);
   335→   335→
   336→   336→  // Find the max height for normalization
   337→   337→  let maxH = 0;
   338→   338→  for (let i = 0; i < vertexCount; i++) {
   339→   339→    if (heights[i] > maxH) maxH = heights[i];
   340→   340→  }
   341→   341→
   342→   342→  for (let i = 0; i < vertexCount; i++) {
   343→   343→    const y = posArr[i * 3 + 1];
   344→   344→    const isTop = y >= topThreshold - 0.01;
   345→   345→
   346→   346→    if (isTop && maxH > 0) {
   347→   347→      const t = heights[i] / maxH;
   348→   348→      const [r, g, b] = sampleGradient(palette, t);
   349→   349→      colors[i * 3] = r;
   350→   350→      colors[i * 3 + 1] = g;
   351→   351→      colors[i * 3 + 2] = b;
   352→   352→    } else {
   353→   353→      // Side faces: darken version of the low color
   354→   354→      colors[i * 3] = palette.lowColor[0] * 0.6;
   355→   355→      colors[i * 3 + 1] = palette.lowColor[1] * 0.6;
   356→   356→      colors[i * 3 + 2] = palette.lowColor[2] * 0.6;
   357→   357→    }
   358→   358→  }
   359→   359→
   360→   360→  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   361→   361→  return geo;
   362→   362→}

   749→   749→function TerritoryMesh({ territoryId, path, labelX, labelY, onHover, onClick }: TerritoryMeshProps) {
   750→   750→  const territories = useGameStore(s => s.territories);
   751→   751→  const selectedTerritory = useGameStore(s => s.selectedTerritory);
   752→   752→  const targetTerritory = useGameStore(s => s.targetTerritory);
   753→   753→  const phase = useGameStore(s => s.phase);
   754→   754→  const players = useGameStore(s => s.players);
   755→   755→  const reinforcementsLeft = useGameStore(s => s.reinforcementsLeft);
   756→   756→  const deployUnitType = useGameStore(s => s.deployUnitType);
   757→   757→  const currentPlayerIndex = useGameStore(s => s.currentPlayerIndex);
   758→   758→  const selectTerritory = useGameStore(s => s.selectTerritory);
   759→   759→  const deployArmy = useGameStore(s => s.deployArmy);
   760→   760→  const battleAnimation = useGameStore(s => s.battleAnimation);
   761→   761→
   762→   762→  const territory = territories[territoryId];
   763→   763→  if (!territory) return null;
   764→   764→
   765→   765→  const visual = TERRITORY_VISUAL[territoryId] || { structure: 'tower' as StructureType, height: 0.2, structureColor: '#8B7355', terrainColor: '#5A5A3A', terrainSideColor: '#3A3A2A' };
   766→   766→  const playerColorMap = useMemo(() => {
   767→   767→    const map: Record<string, { color: string; colorLight: string }> = {};
   768→   768→    players.forEach(p => { map[p.id] = { color: p.color, colorLight: p.colorLight }; });
   769→   769→    return map;
   770→   770→  }, [players]);
   771→   771→
   772→   772→  const owner = territory.ownerId ? playerColorMap[territory.ownerId] : null;
   773→   773→  const currentPlayerId = phase !== 'setup' && phase !== 'gameover' ? players[currentPlayerIndex]?.id : null;
   774→   774→
   775→   775→  const isSelected = territoryId === selectedTerritory;
   776→   776→  const isTarget = territoryId === targetTerritory;
   777→   777→
   778→   778→  const geometry = useMemo(() => {
   779→   779→    const tDef = TERRITORY_DEFS.find(t => t.id === territoryId);
   780→   780→    const region = tDef?.region || '';
   781→   781→    return createTerritoryGeometry(path, visual.height, territoryId, region);
   782→   782→  }, [path, visual.height, territoryId]);
   783→   783→  const edgesGeometry = useMemo(() => new THREE.EdgesGeometry(geometry, 25), [geometry]);
   784→   784→
   785→   785→  const [cx, cz] = svgTo3D(labelX, labelY);
   786→   786→  const structY = 0.05 + visual.height;
   787→   787→  const dominant = getDominantUnit(territory.units);
   788→   788→
   789→   789→  const isSelectable = useMemo(() => {
   790→   790→    if (phase === 'deploy') {
   791→   791→      const cost = getUnitCost(deployUnitType, players[currentPlayerIndex]?.characterClass || '');
   792→   792→      return territory.ownerId === currentPlayerId && reinforcementsLeft >= cost;
   793→   793→    }
   794→   794→    if (phase === 'attack') {
   795→   795→      if (selectedTerritory === null) return territory.ownerId === currentPlayerId && territory.units.length > 1;
   796→   796→      if (territoryId === selectedTerritory) return true;
   797→   797→      if (territory.ownerId !== currentPlayerId) return territory.adjacentTo.includes(selectedTerritory);
   798→   798→      return territory.ownerId === currentPlayerId && territory.units.length > 1;
   799→   799→    }

  1332→  1332→function BattleExplosion() {
  1333→  1333→  const battleAnimation = useGameStore(s => s.battleAnimation);
  1334→  1334→  const territories = useGameStore(s => s.territories);
  1335→  1335→  const TERRITORY_DEFS_LOCAL = TERRITORY_DEFS;
  1336→  1336→  const pointsRef = useRef<THREE.Points>(null);
  1337→  1337→  const [active, setActive] = useState(false);
  1338→  1338→  const [positions, setPositions] = useState<Float32Array | null>(null);
  1339→  1339→  const [velocities] = useState(() => {
  1340→  1340→    // Pre-compute random velocities for 40 particles
  1341→  1341→    const vels = new Float32Array(40 * 3);
  1342→  1342→    for (let i = 0; i < 40; i++) {
  1343→  1343→      const angle = Math.random() * Math.PI * 2;
  1344→  1344→      const speed = 0.5 + Math.random() * 2;
  1345→  1345→      const upSpeed = 1 + Math.random() * 3;
  1346→  1346→      vels[i * 3] = Math.cos(angle) * speed;
  1347→  1347→      vels[i * 3 + 1] = upSpeed;
  1348→  1348→      vels[i * 3 + 2] = Math.sin(angle) * speed;
  1349→  1349→    }
  1350→  1350→    return vels;
  1351→  1351→  });
  1352→  1352→  const startTimeRef = useRef(0);
  1353→  1353→
  1354→  1354→  useEffect(() => {
  1355→  1355→    if (battleAnimation) {
  1356→  1356→      const targetDef = TERRITORY_DEFS_LOCAL.find(t => t.id === battleAnimation.toTerritory);
  1357→  1357→      if (!targetDef) return;
  1358→  1358→      const [cx, cz] = svgTo3D(targetDef.labelX, targetDef.labelY);
  1359→  1359→      const height = (TERRITORY_VISUAL[battleAnimation.toTerritory]?.height || 0.2) + 0.3;
  1360→  1360→
  1361→  1361→      // Initialize particle positions at the target location
  1362→  1362→      const pos = new Float32Array(40 * 3);
  1363→  1363→      for (let i = 0; i < 40; i++) {
  1364→  1364→        pos[i * 3] = cx + (Math.random() - 0.5) * 0.3;
  1365→  1365→        pos[i * 3 + 1] = height;
  1366→  1366→        pos[i * 3 + 2] = cz + (Math.random() - 0.5) * 0.3;
  1367→  1367→      }
  1368→  1368→      setPositions(pos);
  1369→  1369→      setActive(true);
  1370→  1370→      startTimeRef.current = Date.now();
  1371→  1371→    }
  1372→  1372→  }, [battleAnimation, territories, TERRITORY_DEFS_LOCAL]);
  1373→  1373→
  1374→  1374→  useFrame(() => {
  1375→  1375→    if (!active || !pointsRef.current || !positions) return;
  1376→  1376→    const age = (Date.now() - startTimeRef.current) / 1000;
  1377→  1377→    if (age > 1.5) {
  1378→  1378→      setActive(false);
  1379→  1379→      return;
  1380→  1380→    }
  1381→  1381→
  1382→  1382→    const posArr = pointsRef.current.geometry.attributes.position.array as Float32Array;
  1383→  1383→    for (let i = 0; i < 40; i++) {
  1384→  1384→      const t = age;
  1385→  1385→      posArr[i * 3] = posArr[i * 3] + velocities[i * 3] * 0.008;
  1386→  1386→      posArr[i * 3 + 1] = posArr[i * 3 + 1] + velocities[i * 3 + 1] * 0.008 - t * t * 2 * 0.008; // gravity
  1387→  1387→      posArr[i * 3 + 2] = posArr[i * 3 + 2] + velocities[i * 3 + 2] * 0.008;
  1388→  1388→    }
  1389→  1389→    pointsRef.current.geometry.attributes.position.needsUpdate = true;
  1390→  1390→
  1391→  1391→    // Fade out
  1392→  1392→    const mat = pointsRef.current.material as THREE.PointsMaterial;
  1393→  1393→    mat.opacity = Math.max(0, 1 - age / 1.5);
  1394→  1394→    mat.size = 0.12 * (1 + age * 0.5);
  1395→  1395→  });
  1396→  1396→
  1397→  1397→  if (!active || !positions) return null;
  1398→  1398→
  1399→  1399→  return (
  1400→  1400→    <points ref={pointsRef}>
  1401→  1401→      <bufferGeometry>
  1402→  1402→        <bufferAttribute
  1403→  1403→          attach="attributes-position"
  1404→  1404→          args={[positions, 3]}
  1405→  1405→        />
  1406→  1406→      </bufferGeometry>
  1407→  1407→      <pointsMaterial
  1408→  1408→        size={0.12}
  1409→  1409→        color={battleAnimation?.conquered ? '#FF8800' : '#FF3333'}
  1410→  1410→        transparent
  1411→  1411→        opacity={1}
  1412→  1412→        sizeAttenuation
  1413→  1413→        blending={THREE.AdditiveBlending}
  1414→  1414→        depthWrite={false}
  1415→  1415→      />
  1416→  1416→    </points>
  1417→  1417→  );
  1418→  1418→}

   141→   141→function randomlyAssignTerritories(playerCount: number, players: Player[]): Record<string, TerritoryState> {
   142→   142→  const territories: Record<string, TerritoryState> = {};
   143→   143→  const territoryIds = TERRITORIES.map(t => t.id);
   144→   144→  // Shuffle
   145→   145→  for (let i = territoryIds.length - 1; i > 0; i--) {
   146→   146→    const j = Math.floor(Math.random() * (i + 1));
   147→   147→    [territoryIds[i], territoryIds[j]] = [territoryIds[j], territoryIds[i]];
   148→   148→  }
   149→   149→
   150→   150→  // Default unit types to distribute
   151→   151→  const defaultUnits: UnitTypeId[] = ['spearman', 'swordsman', 'swordsman', 'archer', 'cavalry', 'shield_bearer', 'mage', 'assassin'];
   152→   152→
   153→   153→  for (let i = 0; i < territoryIds.length; i++) {
   154→   154→    const def = TERRITORIES.find(t => t.id === territoryIds[i])!;
   155→   155→    const playerIndex = i % playerCount;
   156→   156→    const unitType = defaultUnits[i % defaultUnits.length];
   157→   157→    territories[territoryIds[i]] = {
   158→   158→      ...def,
   159→   159→      ownerId: players[playerIndex].id,
   160→   160→      units: [unitType], // Start with 1 unit each
   161→   161→    };
   162→   162→  }
   163→   163→
   164→   164→  // Give each player bonus units to reach minimum 2-3 units per territory on average
   165→   165→  const territoriesPerPlayer = Math.floor(16 / playerCount);
   166→   166→  for (const player of players) {
   167→   167→    const playerTerritories = Object.values(territories).filter(t => t.ownerId === player.id);
   168→   168→    const bonusUnits = Math.max(0, Math.ceil(territoriesPerPlayer / 2));
   169→   169→    for (let i = 0; i < bonusUnits; i++) {
   170→   170→      const randomTerritory = playerTerritories[Math.floor(Math.random() * playerTerritories.length)];
   171→   171→      const randomUnit = defaultUnits[Math.floor(Math.random() * defaultUnits.length)];
   172→   172→      territories[randomTerritory.id].units.push(randomUnit);
   173→   173→    }
   174→   174→  }
   175→   175→
   176→   176→  return territories;
   177→   177→}

   425→   425→
   426→   426→      const oldOwner = newTerritories[state.targetTerritory].ownerId;
   427→   427→      const newOwner = newTerritories[state.selectedTerritory].ownerId;
   428→   428→
   429→   429→      // Move top units to conquered territory
   430→   430→      const remaining = newTerritories[state.selectedTerritory].units;
   431→   431→      const movedUnits = remaining.slice(0, movingUnits);
   432→   432→      const stayedUnits = remaining.slice(movingUnits);
   433→   433→
   434→   434→      newTerritories[state.targetTerritory] = {
   435→   435→        ...newTerritories[state.targetTerritory],
   436→   436→        ownerId: newOwner,
   437→   437→        units: movedUnits,
   438→   438→      };
   439→   439→      newTerritories[state.selectedTerritory] = {
   440→   440→        ...newTerritories[state.selectedTerritory],
   441→   441→        units: stayedUnits,
   442→   442→      };

   444→   444→      if (oldOwner) {

   446→   446→        oldPlayer.territories = oldPlayer.territories.filter(id => id !== state.targetTerritory);
   447→   447→        if (oldPlayer.territories.length === 0) {
   448→   448→          oldPlayer.eliminated = true;
   449→   449→          newLog.push(addLog(state, `💀 ${oldPlayer.name} has been eliminated!`, 'info'));
   450→   450→
   451→   451→          // Queue elimination story for AI players
   452→   452→          if (oldPlayer.isAI) {
   453→   453→            const elimClass = oldPlayer.characterClass.toLowerCase();
   454→   454→            const elimConfig = PLAYER_CONFIGS.find(p => p.characterClass.toLowerCase() === elimClass) || PLAYER_CONFIGS[0];
   455→   455→            const elimBeat = getEliminationBeat(oldPlayer.name, elimClass, oldPlayer.color);
   456→   456→            get().queueStory(elimBeat);
   457→   457→          }
   458→   458→        }
   459→   459→      }
   460→   460→      const newOwnerPlayer = newPlayers.find(p => p.id === newOwner)!;

   462→   462→
   463→   463→      newLog.push(addLog(state, `🏆 ${attacker.name} conquered ${toTerritory.name} from ${defender.name}!`, 'conquer'));
